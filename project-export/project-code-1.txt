import { NextResponse } from 'next/server';
import { createAdminSupabase } from '@/lib/supabase/admin';

export async function POST(request: Request) {
  try {
    const body = await request.json();
    const fullName = String(body.fullName || '').trim();
    const email = String(body.email || '').trim().toLowerCase();
    const password = String(body.password || '');
    const role = body.role === 'teacher' ? 'teacher' : 'student';
    const inviteCode = String(body.inviteCode || '').trim();
    const classCode = String(body.classCode || '').trim().toUpperCase();

    if (!fullName || !email || password.length < 6) {
      return NextResponse.json(
        { error: 'Заполните имя, email и пароль (минимум 6 символов).' },
        { status: 400 }
      );
    }

    const admin = createAdminSupabase();
    let classroomId: string | null = null;

    if (role === 'teacher') {
      const secret = process.env.TEACHER_INVITE_CODE;
      if (!secret) {
        return NextResponse.json(
          { error: 'На сервере не настроен код приглашения для учителей.' },
          { status: 500 }
        );
      }
      if (inviteCode !== secret) {
        return NextResponse.json(
          { error: 'Неверный код приглашения учителя.' },
          { status: 403 }
        );
      }
    } else if (classCode) {
      const { data: cls } = await admin
        .from('classrooms')
        .select('id')
        .eq('join_code', classCode)
        .maybeSingle();

      if (!cls) {
        return NextResponse.json(
          { error: `Класс с кодом «${classCode}» не найден. Проверьте код у учителя.` },
          { status: 400 }
        );
      }
      classroomId = cls.id;
    }

    const { data, error } = await admin.auth.admin.createUser({
      email,
      password,
      email_confirm: true,
      user_metadata: { full_name: fullName },
    });

    if (error || !data.user) {
      return NextResponse.json(
        { error: error?.message || 'Не удалось создать аккаунт.' },
        { status: 400 }
      );
    }

    const { error: profErr } = await admin.from('profiles').upsert({
      id: data.user.id,
      full_name: fullName,
      role,
      classroom_id: classroomId,
    });

    if (profErr) {
      return NextResponse.json({ error: profErr.message }, { status: 500 });
    }

    return NextResponse.json({ ok: true, role });
  } catch (err) {
    console.error('auth/register error:', err);
    return NextResponse.json(
      { error: 'Внутренняя ошибка при регистрации.' },
      { status: 500 }
    );
  }
}

===== src/app/api/homework/check/route.ts =====
import { NextResponse } from 'next/server';
import { createServerSupabase } from '@/lib/supabase/server';
import { createAdminSupabase } from '@/lib/supabase/admin';

function normalizeAnswer(val: string): string {
  return val.trim().replace(/\s+/g, '').replace(/\./g, ',').toLowerCase();
}

export async function POST(request: Request) {
  try {
    const supabase = await createServerSupabase();
    const {
      data: { user },
    } = await supabase.auth.getUser();

    if (!user) {
      return NextResponse.json({ error: 'Нужно войти в аккаунт' }, { status: 401 });
    }

    const body = await request.json();
    const taskId = String(body.taskId || '');
    const answerText = String(body.answerText || '');
    const photoPath = body.photoPath ? String(body.photoPath) : null;

    if (!taskId) {
      return NextResponse.json({ error: 'Не указан ID задания' }, { status: 400 });
    }

    const admin = createAdminSupabase();

    const { data: task, error: taskErr } = await admin
      .from('homework_tasks')
      .select('id, type, points')
      .eq('id', taskId)
      .single();

    if (taskErr || !task) {
      return NextResponse.json({ error: 'Задание не найдено' }, { status: 404 });
    }

    // Развёрнутая задача отправляется на ручную проверку учителю
    if (task.type === 'open') {
      if (!answerText.trim() && !photoPath) {
        return NextResponse.json(
          { error: 'Напишите решение или прикрепите фото тетради' },
          { status: 400 }
        );
      }

      const { data: existing } = await admin
        .from('submissions')
        .select('photo_path')
        .eq('task_id', task.id)
        .eq('student_id', user.id)
        .maybeSingle();

      const { data: submission, error: subErr } = await admin
        .from('submissions')
        .upsert(
          {
            task_id: task.id,
            student_id: user.id,
            answer_text: answerText.trim(),
            photo_path: photoPath ?? existing?.photo_path ?? null,
            is_correct: null,
            score: 0,
            status: 'pending_review',
            updated_at: new Date().toISOString(),
          },
          { onConflict: 'task_id,student_id' }
        )
        .select()
        .single();

      if (subErr) {
        return NextResponse.json({ error: subErr.message }, { status: 500 });
      }

      return NextResponse.json({ submission, hint: '' });
    }

    // Автопроверка для test и input
    const { data: keyRow, error: keyErr } = await admin
      .from('homework_answer_keys')
      .select('correct_answer, solution_hint')
      .eq('task_id', task.id)
      .single();

    if (keyErr || !keyRow) {
      return NextResponse.json(
        { error: 'Для этого задания не задан ключ ответа' },
        { status: 500 }
      );
    }

    const isCorrect =
      normalizeAnswer(answerText) === normalizeAnswer(keyRow.correct_answer);
    const score = isCorrect ? task.points : 0;

    const { data: submission, error: subErr } = await admin
      .from('submissions')
      .upsert(
        {
          task_id: task.id,
          student_id: user.id,
          answer_text: answerText.trim(),
          is_correct: isCorrect,
          score,
          status: 'checked',
          updated_at: new Date().toISOString(),
        },
        { onConflict: 'task_id,student_id' }
      )
      .select()
      .single();

    if (subErr) {
      return NextResponse.json({ error: subErr.message }, { status: 500 });
    }

    return NextResponse.json({
      submission,
      hint: isCorrect ? '' : keyRow.solution_hint || '',
    });
  } catch (err) {
    console.error('homework/check error:', err);
    return NextResponse.json({ error: 'Внутренняя ошибка сервера' }, { status: 500 });
  }
}

===== src/app/api/simulators/progress/route.ts =====
import { NextResponse } from 'next/server';
import { createServerSupabase } from '@/lib/supabase/server';
import { createAdminSupabase } from '@/lib/supabase/admin';

export async function POST(request: Request) {
  try {
    const supabase = await createServerSupabase();
    const {
      data: { user },
    } = await supabase.auth.getUser();

    // Гости могут тренироваться без сохранения в БД
    if (!user) {
      return NextResponse.json({ ok: true, guest: true });
    }

    const body = await request.json();
    const simId = String(body.simId || 'generator').trim();
    const isCorrect = Boolean(body.isCorrect);

    const admin = createAdminSupabase();

    // Ищем тему с подходящим симулятором (или первую тему курса как запасной вариант)
    const targetSimType = simId === 'generator' ? 'arithmetic' : simId;
    let { data: topic } = await admin
      .from('topics')
      .select('id')
      .eq('sim_type', targetSimType)
      .order('order_index', { ascending: true })
      .limit(1)
      .maybeSingle();

    if (!topic) {
      const { data: fallback } = await admin
        .from('topics')
        .select('id')
        .order('order_index', { ascending: true })
        .limit(1)
        .maybeSingle();
      topic = fallback;
    }

    if (!topic) {
      return NextResponse.json({ ok: true, skipped: true });
    }

    const { data: existing } = await admin
      .from('topic_progress')
      .select('completed, sim_correct_count, sim_streak')
      .eq('student_id', user.id)
      .eq('topic_id', topic.id)
      .maybeSingle();

    const nextCorrect =
      Number(existing?.sim_correct_count || 0) + (isCorrect ? 1 : 0);
    const nextStreak = isCorrect ? Number(existing?.sim_streak || 0) + 1 : 0;

    const { error } = await admin.from('topic_progress').upsert({
      student_id: user.id,
      topic_id: topic.id,
      completed: Boolean(existing?.completed),
      sim_correct_count: nextCorrect,
      sim_streak: nextStreak,
      updated_at: new Date().toISOString(),
    });

    if (error) {
      return NextResponse.json({ error: error.message }, { status: 500 });
    }

    return NextResponse.json({
      ok: true,
      topicId: topic.id,
      nextCorrect,
      nextStreak,
    });
  } catch (err) {
    console.error('simulators/progress error:', err);
    return NextResponse.json(
      { error: 'Ошибка сохранения прогресса' },
      { status: 500 }
    );
  }
}

import { NextResponse } from 'next/server';
import { createServerSupabase } from '@/lib/supabase/server';
import { createAdminSupabase } from '@/lib/supabase/admin';
import {
  BADGES,
  PRACTICE_STAR_TARGET,
  XP_REWARDS,
  getAlmatyDateStr,
  getYesterdayAlmatyDateStr,
  type StudentGamificationStats,
  type TopicStarProgress,
} from '@/lib/gamification';

type GamificationAction = 'slides_done' | 'practice_solved' | 'homework_done';

export async function GET() {
  try {
    const supabase = await createServerSupabase();
    const {
      data: { user },
    } = await supabase.auth.getUser();

    if (!user) {
      return NextResponse.json({ error: 'Нужен вход в аккаунт' }, { status: 401 });
    }

    const admin = createAdminSupabase();
    const [{ data: profile }, { data: rows }] = await Promise.all([
      admin
        .from('profiles')
        .select('xp, streak_days, best_streak, last_active_date, badges')
        .eq('id', user.id)
        .single(),
      admin
        .from('lesson_stars')
        .select('topic_id, slides_done, practice_done, practice_solved, homework_done')
        .eq('user_id', user.id),
    ]);

    const today = getAlmatyDateStr();
    const yesterday = getYesterdayAlmatyDateStr();
    const lastActive = profile?.last_active_date
      ? String(profile.last_active_date).slice(0, 10)
      : null;

    // Если последний раз заходил раньше вчерашнего дня — текущий стрик обнулился
    const liveStreak =
      lastActive === today || lastActive === yesterday
        ? Number(profile?.streak_days || 0)
        : 0;

    const stats: StudentGamificationStats = {
      xp: Number(profile?.xp || 0),
      streak_days: liveStreak,
      best_streak: Number(profile?.best_streak || 0),
      last_active_date: lastActive,
      badges: Array.isArray(profile?.badges) ? profile.badges : [],
    };

    const starsMap: Record<string, TopicStarProgress> = {};
    for (const r of rows || []) {
      const tid = String(r.topic_id);
      starsMap[tid] = {
        topic_id: tid,
        slides_done: Boolean(r.slides_done),
        practice_done: Boolean(r.practice_done),
        practice_solved: Number(r.practice_solved || 0),
        homework_done: Boolean(r.homework_done),
      };
    }

    return NextResponse.json({ stats, starsMap });
  } catch (err) {
    console.error('GET /api/student/gamification error:', err);
    return NextResponse.json(
      { error: 'Не удалось загрузить прогресс' },
      { status: 500 }
    );
  }
}

export async function POST(request: Request) {
  try {
    const supabase = await createServerSupabase();
    const {
      data: { user },
    } = await supabase.auth.getUser();

    if (!user) {
      return NextResponse.json({ error: 'Нужен вход в аккаунт' }, { status: 401 });
    }

    const body = await request.json();
    const topicId = String(body.topicId || '').trim();
    const action = body.action as GamificationAction;
    const solvedDelta = Math.max(0, Math.min(10, Number(body.solvedDelta ?? 1)));
    const solvedTotal =
      body.solvedTotal !== undefined
        ? Math.max(0, Math.min(100, Number(body.solvedTotal)))
        : undefined;

    if (
      !topicId ||
      !['slides_done', 'practice_solved', 'homework_done'].includes(action)
    ) {
      return NextResponse.json(
        { error: 'Некорректные параметры прогресса' },
        { status: 400 }
      );
    }

    const admin = createAdminSupabase();

    const [{ data: existing }, { data: profile }] = await Promise.all([
      admin
        .from('lesson_stars')
        .select('topic_id, slides_done, practice_done, practice_solved, homework_done')
        .eq('user_id', user.id)
        .eq('topic_id', topicId)
        .maybeSingle(),
      admin
        .from('profiles')
        .select('xp, streak_days, best_streak, last_active_date, badges')
        .eq('id', user.id)
        .single(),
    ]);

    const prev: TopicStarProgress = {
      topic_id: topicId,
      slides_done: Boolean(existing?.slides_done),
      practice_done: Boolean(existing?.practice_done),
      practice_solved: Number(existing?.practice_solved || 0),
      homework_done: Boolean(existing?.homework_done),
    };

    const next: TopicStarProgress = { ...prev };
    let awardedXp = 0;

    if (action === 'slides_done' && !prev.slides_done) {
      next.slides_done = true;
      awardedXp += XP_REWARDS.slides_done;
    }

    if (action === 'practice_solved') {
      const updatedSolved =
        solvedTotal !== undefined
          ? Math.max(prev.practice_solved, solvedTotal)
          : prev.practice_solved + solvedDelta;
      next.practice_solved = updatedSolved;

      if (!prev.practice_done && updatedSolved >= PRACTICE_STAR_TARGET) {
        next.practice_done = true;
        awardedXp += XP_REWARDS.practice_done;
      }
    }

    if (action === 'homework_done' && !prev.homework_done) {
      next.homework_done = true;
      awardedXp += XP_REWARDS.homework_done;
    }

    const wasPerfect = prev.slides_done && prev.practice_done && prev.homework_done;
    const isPerfect = next.slides_done && next.practice_done && next.homework_done;
    if (!wasPerfect && isPerfect) {
      awardedXp += XP_REWARDS.perfect_lesson_bonus;
    }

    const { error: upsertErr } = await admin.from('lesson_stars').upsert(
      {
        user_id: user.id,
        topic_id: topicId,
        slides_done: next.slides_done,
        practice_done: next.practice_done,
        practice_solved: next.practice_solved,
        homework_done: next.homework_done,
        updated_at: new Date().toISOString(),
      },
      { onConflict: 'user_id,topic_id' }
    );

    if (upsertErr) {
      console.error('lesson_stars upsert error:', upsertErr);
      return NextResponse.json(
        { error: 'Не удалось сохранить звёзды урока' },
        { status: 500 }
      );
    }

    // Обновляем серию дней (стрик) по часовому поясу Казахстана
    const today = getAlmatyDateStr();
    const yesterday = getYesterdayAlmatyDateStr();
    const lastActive = profile?.last_active_date
      ? String(profile.last_active_date).slice(0, 10)
      : null;

    let nextStreak = Number(profile?.streak_days || 0);
    if (lastActive === today) {
      nextStreak = Math.max(1, nextStreak);
    } else if (lastActive === yesterday) {
      nextStreak = nextStreak + 1;
    } else {
      nextStreak = 1;
    }

    const nextBestStreak = Math.max(Number(profile?.best_streak || 0), nextStreak);
    const nextXp = Number(profile?.xp || 0) + awardedXp;

    // Считаем все достижения ученика
    const { data: allRows } = await admin
      .from('lesson_stars')
      .select('slides_done, practice_done, practice_solved, homework_done')
      .eq('user_id', user.id);

    const rows = allRows || [];
    const totalStars = rows.reduce(
      (acc, r) =>
        acc +
        (r.slides_done ? 1 : 0) +
        (r.practice_done ? 1 : 0) +
        (r.homework_done ? 1 : 0),
      0
    );
    const perfectLessons = rows.filter(
      (r) => r.slides_done && r.practice_done && r.homework_done
    ).length;
    const totalSlidesDone = rows.filter((r) => r.slides_done).length;
    const totalPracticeSolved = rows.reduce(
      (acc, r) => acc + Number(r.practice_solved || 0),
      0
    );

    const prevBadges = new Set<string>(
      Array.isArray(profile?.badges) ? profile.badges : []
    );
    const nextBadges = new Set<string>(prevBadges);

    if (totalStars >= 1) nextBadges.add('first_star');
    if (perfectLessons >= 1) nextBadges.add('perfect_lesson');
    if (totalPracticeSolved >= 15) nextBadges.add('practice_fan');
    if (totalSlidesDone >= 5) nextBadges.add('slides_reader');
    if (nextStreak >= 3) nextBadges.add('streak_3');
    if (nextStreak >= 7) nextBadges.add('streak_7');
    if (nextXp >= 500) nextBadges.add('xp_500');
    if (perfectLessons >= 5) nextBadges.add('lessons_5_perfect');

    const newBadgeIds = Array.from(nextBadges).filter((b) => !prevBadges.has(b));
    const newBadges = BADGES.filter((b) => newBadgeIds.includes(b.id));

    await admin
      .from('profiles')
      .update({
        xp: nextXp,
        streak_days: nextStreak,
        best_streak: nextBestStreak,
        last_active_date: today,
        badges: Array.from(nextBadges),
      })
      .eq('id', user.id);

    const stats: StudentGamificationStats = {
      xp: nextXp,
      streak_days: nextStreak,
      best_streak: nextBestStreak,
      last_active_date: today,
      badges: Array.from(nextBadges),
    };

    return NextResponse.json({
      stats,
      topicStars: next,
      awardedXp,
      newBadges,
    });
  } catch (err) {
    console.error('POST /api/student/gamification error:', err);
    return NextResponse.json(
      { error: 'Ошибка сохранения прогресса' },
      { status: 500 }
    );
  }
}

import { NextResponse } from 'next/server';
import { createServerSupabase } from '@/lib/supabase/server';
import { createAdminSupabase } from '@/lib/supabase/admin';

export const maxDuration = 60;

const MODELS = [
  'gemini-4-hydrogen',
  'gemini-3.8-flash',
  'gemini-3.6-flash',
  'gemini-3.5-flash-lite',
  'gemini-3.1-flash-lite',
];

const REVIEW_SCHEMA = {
  type: 'OBJECT',
  properties: {
    transcribed_solution: { type: 'STRING' },
    verdict: {
      type: 'STRING',
      enum: ['correct', 'partial', 'incorrect', 'unreadable'],
    },
    suggested_score: { type: 'INTEGER' },
    error_step: { type: 'STRING' },
    teacher_comment: { type: 'STRING' },
  },
  required: [
    'transcribed_solution',
    'verdict',
    'suggested_score',
    'error_step',
    'teacher_comment',
  ],
};

interface GeminiPart {
  text?: string;
  thought?: boolean;
  inlineData?: {
    mimeType: string;
    data: string;
  };
}

interface AiReviewResult {
  transcribed_solution: string;
  verdict: 'correct' | 'partial' | 'incorrect' | 'unreadable';
  suggested_score: number;
  error_step: string;
  teacher_comment: string;
}

function inferMimeType(path: string, blobType?: string): string {
  if (blobType && blobType.startsWith('image/')) return blobType;
  const lower = path.toLowerCase();
  if (lower.endsWith('.png')) return 'image/png';
  if (lower.endsWith('.webp')) return 'image/webp';
  if (lower.endsWith('.gif')) return 'image/gif';
  return 'image/jpeg';
}

function extractAnswerText(data: {
  candidates?: { content?: { parts?: GeminiPart[] } }[];
}): string {
  const parts = data?.candidates?.[0]?.content?.parts || [];
  const finalParts = parts.filter(
    (p) => !p.thought && typeof p.text === 'string' && p.text.trim().length > 0
  );
  if (finalParts.length > 0) {
    return finalParts.map((p) => p.text).join('\n');
  }
  return parts[parts.length - 1]?.text || '';
}

function parseReviewJson(rawText: string, maxPoints: number): AiReviewResult | null {
  const cleaned = rawText
    .replace(/```json/gi, '')
    .replace(/```/g, '')
    .trim();

  let parsed: unknown = null;
  try {
    parsed = JSON.parse(cleaned);
  } catch {
    const start = cleaned.indexOf('{');
    const end = cleaned.lastIndexOf('}');
    if (start !== -1 && end > start) {
      const slice = cleaned.slice(start, end + 1);
      try {
        parsed = JSON.parse(slice);
      } catch {
        try {
          parsed = JSON.parse(slice.replace(/\\(?!["\\/bfnrtu])/g, '\\\\'));
        } catch {
          parsed = null;
        }
      }
    }
  }

  const obj = parsed as Record<string, unknown>;

    rawVerdict === 'correct' ||
    rawVerdict === 'partial' ||
    rawVerdict === 'incorrect' ||
    rawVerdict === 'unreadable'
      ? rawVerdict
      : 'partial';

  const rawScore = Number(obj.suggested_score);
  const clampedScore = Number.isFinite(rawScore)
    ? Math.max(0, Math.min(maxPoints, Math.round(rawScore)))
    : 0;

  return {
    transcribed_solution: String(obj.transcribed_solution || '').trim(),
    verdict,
    suggested_score: clampedScore,
    error_step: String(obj.error_step || '').trim(),
    teacher_comment: String(obj.teacher_comment || '').trim(),
  };
}

interface GeminiPart {

function extractAnswerText(data: {
  candidates?: { content?: { parts?: GeminiPart[] } }[];
}): string {
  const parts = data?.candidates?.[0]?.content?.parts || [];
  const finalParts = parts.filter(
    (p) => !p.thought && typeof p.text === 'string' && p.text.trim().length > 0
  );
  if (finalParts.length > 0) {
    return finalParts.map((p) => p.text).join('\n');
  }
  return parts[parts.length - 1]?.text || '';
}

function tryParseJson(chunk: string): unknown | null {
  try {
    return JSON.parse(chunk);
  } catch {
    try {
      const repaired = chunk.replace(/\\(?!["\\/bfnrtu])/g, '\\\\');
      return JSON.parse(repaired);
    } catch {
      return null;
    }
  }
}

function extractJsonValues(rawText: string): unknown[] {
  const cleaned = rawText
    .replace(/```json/gi, '')
    .replace(/```/g, '')
    .trim();

  const direct = tryParseJson(cleaned);
  if (direct !== null) return [direct];

  const values: unknown[] = [];
  let i = 0;

  while (i < cleaned.length) {
    const ch = cleaned[i];
    if (ch !== '{' && ch !== '[') {
      i++;
      continue;
    }

    const openChar = ch;
    const closeChar = ch === '{' ? '}' : ']';
    let depth = 0;
    let inString = false;
    let escaped = false;
    let end = -1;

    for (let j = i; j < cleaned.length; j++) {
      const c = cleaned[j];
      if (inString) {
        if (escaped) {
          escaped = false;
        } else if (c === '\\') {
          escaped = true;
        } else if (c === '"') {
          inString = false;
        }
        continue;
      }

      if (c === '"') {
        inString = true;
      } else if (c === openChar) {
        depth++;
      } else if (c === closeChar) {
        depth--;
        if (depth === 0) {
          end = j;
          break;
        }
      }
    }

    if (end !== -1) {
      const slice = cleaned.slice(i, end + 1);
      const parsed = tryParseJson(slice);
      if (parsed !== null) {
        values.push(parsed);
        i = end + 1;
        continue;
      }
    }

    i++;
  }

  return values;
}

function collectRawTasks(values: unknown[]): unknown[] {
  const list: unknown[] = [];
  for (const val of values) {
    if (Array.isArray(val)) {
      list.push(...val);
    } else if (val && typeof val === 'object') {
      const obj = val as Record<string, unknown>;
      if (Array.isArray(obj.tasks)) {
        list.push(...obj.tasks);
      } else if (typeof obj.question_md === 'string') {
        list.push(obj);
      }
    }
  }
  return list;
}

function sanitizeTasks(raw: unknown): DraftGenTask[] {
  if (!Array.isArray(raw)) return [];
  const tasks: DraftGenTask[] = [];

  for (const item of raw) {
    if (!item || typeof item !== 'object') continue;
    const obj = item as Record<string, unknown>;

    const rawType = String(obj.type || 'test');
    const type: DraftGenTask['type'] =
      rawType === 'input' || rawType === 'open' ? rawType : 'test';

    const rawDiff = String(obj.difficulty || 'standard');
    const difficulty: DraftGenTask['difficulty'] =
      rawDiff === 'base' || rawDiff === 'pro' ? rawDiff : 'standard';

    const question_md = String(obj.question_md || '').trim();
    if (!question_md) continue;

    const options =
      type === 'test' && Array.isArray(obj.options)
        ? obj.options.map((o) => String(o).trim()).filter(Boolean).slice(0, 4)
        : [];

    let correct_answer =
      type === 'open' ? '' : String(obj.correct_answer ?? '').trim();

    if (
      type === 'test' &&
      options.length >= 2 &&
      !options.includes(correct_answer)
    ) {
      correct_answer = options[0];
    }

    const rawPoints = Number(obj.points);
    const points =
      Number.isFinite(rawPoints) && rawPoints >= 1 && rawPoints <= 10
        ? Math.round(rawPoints)
        : difficulty === 'base'
          ? 1
          : difficulty === 'pro'
            ? 3
            : 2;

    tasks.push({
      type: type === 'test' && options.length < 2 ? 'input' : type,
      difficulty,
      points,
      question_md,
      options: type === 'test' && options.length >= 2 ? options : [],
      correct_answer,
      solution_hint: String(obj.solution_hint || '').trim(),
    });
  }

  return tasks;
}

import { NextResponse } from 'next/server';
import { createServerSupabase } from '@/lib/supabase/server';

export const maxDuration = 60;

const VALID_TYPES: SlideType[] = [
  'intro',
  'concept',
  'example',
  'trap',
  'quiz',
  'summary',
];

const MODELS = [
  'gemini-4-hydrogen',
  'gemini-3.8-flash',
  'gemini-3.6-flash',
  'gemini-3.5-flash-lite',
  'gemini-3.1-flash-lite',
];

interface GeminiPart {
  text?: string;
  thought?: boolean;
}

function extractAnswerText(data: {
  candidates?: { content?: { parts?: GeminiPart[] } }[];
}): string {
  const parts = data?.candidates?.[0]?.content?.parts || [];
  const finalParts = parts.filter(
    (p) => !p.thought && typeof p.text === 'string' && p.text.trim().length > 0
  );
  if (finalParts.length > 0) {
    return finalParts.map((p) => p.text).join('\n');
  }
  return parts[parts.length - 1]?.text || '';
}

function tryParseJson(chunk: string): unknown | null {
  try {
    return JSON.parse(chunk);
  } catch {
    try {
      const repaired = chunk.replace(/\\(?!["\\/bfnrtu])/g, '\\\\');
      return JSON.parse(repaired);
    } catch {
      return null;
    }
  }
}

function extractJsonValues(rawText: string): unknown[] {
  const cleaned = rawText
    .replace(/```json/gi, '')
    .replace(/```/g, '')
    .trim();

  const direct = tryParseJson(cleaned);
  if (direct !== null) return [direct];

  const values: unknown[] = [];
  let i = 0;

  while (i < cleaned.length) {
    const ch = cleaned[i];
    if (ch !== '{' && ch !== '[') {
      i++;
      continue;
    }

    const openChar = ch;
    const closeChar = ch === '{' ? '}' : ']';
    let depth = 0;
    let inString = false;
    let escaped = false;
    let end = -1;

    for (let j = i; j < cleaned.length; j++) {
      const c = cleaned[j];
      if (inString) {
        if (escaped) {
          escaped = false;
        } else if (c === '\\') {
          escaped = true;
        } else if (c === '"') {
          inString = false;
        }
        continue;
      }

      if (c === '"') {
        inString = true;
      } else if (c === openChar) {
        depth++;
      } else if (c === closeChar) {
        depth--;
        if (depth === 0) {
          end = j;
          break;
        }
      }
    }

    if (end !== -1) {
      const slice = cleaned.slice(i, end + 1);
      const parsed = tryParseJson(slice);
      if (parsed !== null) {
        values.push(parsed);
        i = end + 1;
        continue;
      }
    }

    i++;
  }

  return values;
}

function collectRawSlides(values: unknown[]): unknown[] {
  const list: unknown[] = [];
  for (const val of values) {
    if (Array.isArray(val)) {
      list.push(...val);
    } else if (val && typeof val === 'object') {
      const obj = val as Record<string, unknown>;
      if (Array.isArray(obj.slides)) {
        list.push(...obj.slides);
      } else if (typeof obj.title === 'string' && typeof obj.body_md === 'string') {
        list.push(obj);
      }
    }
  }
  return list;
}

function sanitizeVisual(raw: unknown): SlideVisual | null {
  if (!raw || typeof raw !== 'object') return null;
  const v = raw as Record<string, unknown>;
  const kind = String(v.kind || '');

  if (kind === 'formula_hero' && typeof v.formula === 'string' && v.formula.trim()) {
    return {
      kind: 'formula_hero',
      formula: v.formula.trim(),
      caption: typeof v.caption === 'string' ? v.caption.trim() : undefined,
    };
  }
  if (kind === 'fraction_bar') {
    const den = Math.max(1, Math.min(16, Number(v.denominator) || 4));
    const num = Math.max(0, Math.min(den, Number(v.numerator) || 1));
    const hasCompare =
      v.compareDenominator !== undefined && v.compareNumerator !== undefined;
    return {
      kind: 'fraction_bar',
      numerator: num,
      denominator: den,
      label: typeof v.label === 'string' ? v.label.trim() : undefined,
      ...(hasCompare
        ? {
            compareDenominator: Math.max(
              1,
              Math.min(16, Number(v.compareDenominator) || 4)
            ),
            compareNumerator: Math.max(0, Number(v.compareNumerator) || 1),
          }
        : {}),
    };
  }
  if (kind === 'number_line') {
    const min = Number(v.min ?? 0);
    const max = Number(v.max ?? 10);
    const points = Array.isArray(v.points)
      ? v.points
          .filter((p) => p && typeof p === 'object')
          .map((p) => {
            const obj = p as Record<string, unknown>;
            return {
              value: Number(obj.value ?? 0),
              label: String(obj.label ?? obj.value ?? ''),
              highlight: obj.highlight !== false,
            };
          })
          .slice(0, 6)
      : [];
    const jumpObj =
      v.jump && typeof v.jump === 'object'
        ? (v.jump as Record<string, unknown>)
        : null;
    return {
      kind: 'number_line',
      min: Number.isFinite(min) ? min : 0,
      max: Number.isFinite(max) && max > min ? max : min + 10,
      points,
      ...(jumpObj
        ? {
            jump: {
              from: Number(jumpObj.from ?? min),
              to: Number(jumpObj.to ?? max),
              label:
                typeof jumpObj.label === 'string'
                  ? jumpObj.label.trim()
                  : undefined,
            },
          }
        : {}),
    };
  }
  if (kind === 'percent_grid') {
    return {
      kind: 'percent_grid',
      percent: Math.max(0, Math.min(100, Number(v.percent) || 25)),
      label: typeof v.label === 'string' ? v.label.trim() : undefined,
    };
  }
  if (kind === 'angle') {
    return {
      kind: 'angle',
      degrees: Math.max(5, Math.min(355, Number(v.degrees) || 60)),
      label: typeof v.label === 'string' ? v.label.trim() : undefined,
    };
  }
  if (kind === 'venn') {
    return {
      kind: 'venn',
      leftLabel: String(v.leftLabel || 'A'),
      rightLabel: String(v.rightLabel || 'B'),
      leftItems: Array.isArray(v.leftItems) ? v.leftItems.map(String).slice(0, 4) : [],
      rightItems: Array.isArray(v.rightItems) ? v.rightItems.map(String).slice(0, 4) : [],
      bothItems: Array.isArray(v.bothItems) ? v.bothItems.map(String).slice(0, 4) : [],
    };
  }
  if (kind === 'compare') {
    return {
      kind: 'compare',
      leftTitle: String(v.leftTitle || 'Случай 1'),
      leftText: String(v.leftText || ''),
      rightTitle: String(v.rightTitle || 'Случай 2'),
      rightText: String(v.rightText || ''),
    };
  }

  return null;
}

function sanitizeSlides(raw: unknown): LessonSlide[] {
  if (!Array.isArray(raw)) return [];
  const result: LessonSlide[] = [];

  for (let i = 0; i < raw.length; i++) {
    const item = raw[i];
    if (!item || typeof item !== 'object') continue;
    const obj = item as Record<string, unknown>;
    const rawType = String(obj.type || 'concept') as SlideType;
    const type: SlideType = VALID_TYPES.includes(rawType) ? rawType : 'concept';
    const title = String(obj.title || `Слайд ${i + 1}`).trim();
    const body_md = String(obj.body_md || '').trim();

    const slide: LessonSlide = {
      id: `s-${Date.now()}-${i}`,
      type,
      title,
      body_md,
      visual: sanitizeVisual(obj.visual),
    };

    if (Array.isArray(obj.steps) && obj.steps.length > 0) {
      slide.steps = obj.steps.map(String).filter(Boolean).slice(0, 6);
    }
    if (typeof obj.wrong_md === 'string' && obj.wrong_md.trim()) {
      slide.wrong_md = obj.wrong_md.trim();
    }
    if (typeof obj.right_md === 'string' && obj.right_md.trim()) {
      slide.right_md = obj.right_md.trim();
    }
    if (Array.isArray(obj.quiz_options) && obj.quiz_options.length >= 2) {
      slide.quiz_options = obj.quiz_options.map(String).slice(0, 4);
      const cIdx = Number(obj.quiz_correct_index ?? 0);
      slide.quiz_correct_index =
        cIdx >= 0 && cIdx < slide.quiz_options.length ? cIdx : 0;
      if (typeof obj.quiz_explanation === 'string') {
        slide.quiz_explanation = obj.quiz_explanation.trim();
      }
    }
    if (Array.isArray(obj.takeaways) && obj.takeaways.length > 0) {
      slide.takeaways = obj.takeaways.map(String).filter(Boolean).slice(0, 5);
    }

    result.push(slide);
  }

  return result;
}

      return NextResponse.json({ error: 'Доступ только для учителя' }, { status: 403 });
    }

    const body = await request.json();
    const studentId = String(body.studentId || '');

    if (!studentId) {
      return NextResponse.json({ error: 'Не указан ID ученика' }, { status: 400 });
    }

    const { data: target } = await admin
      .from('profiles')
      .select('role, full_name')
      .eq('id', studentId)
      .single();

    if (!target || target.role !== 'student') {
      return NextResponse.json(
        { error: 'Можно сбрасывать пароль только ученикам.' },
        { status: 403 }
      );
    }

    const { data: card } = await admin
      .from('student_access_cards')
      .select('student_id, login_username')
      .eq('student_id', studentId)
      .maybeSingle();

    if (card) {
      const newPin = generatePinCode();
      const { error } = await admin.auth.admin.updateUserById(studentId, {
        password: studentPinToPassword(newPin),
      });

      if (error) {
        return NextResponse.json({ error: error.message }, { status: 500 });
      }

      await admin
        .from('student_access_cards')
        .update({ pin_code: newPin, updated_at: new Date().toISOString() })
        .eq('student_id', studentId);

      return NextResponse.json({
        ok: true,
        isPin: true,
        studentName: target.full_name,
        loginUsername: card.login_username,
        tempPassword: newPin,
      });
    }

    const tempPassword = `math-${Math.floor(1000 + Math.random() * 9000)}`;
    const { error } = await admin.auth.admin.updateUserById(studentId, {
      password: tempPassword,
    });

    if (error) {
      return NextResponse.json({ error: error.message }, { status: 500 });
    }

    return NextResponse.json({
      ok: true,
      isPin: false,
      studentName: target.full_name,
      tempPassword,
    });
  } catch (err) {
    console.error('reset-password error:', err);
    return NextResponse.json({ error: 'Ошибка сброса пароля' }, { status: 500 });
  }
}

const MODE_INSTRUCTIONS: Record<TutorMode, { label: string; prompt: string }> = {

const MODELS = [
  'gemini-4-hydrogen',
  'gemini-3.8-flash',
  'gemini-3.6-flash',
  'gemini-3.5-flash-lite',
  'gemini-3.1-flash-lite',
];

interface GeminiPart {
  text?: string;
  thought?: boolean;
}

function extractReplyText(data: {
  candidates?: { content?: { parts?: GeminiPart[] } }[];
}): string {
  const parts = data?.candidates?.[0]?.content?.parts || [];
  const finalParts = parts.filter(
    (p) => !p.thought && typeof p.text === 'string' && p.text.trim().length > 0
  );
  if (finalParts.length > 0) {
    return finalParts
      .map((p) => p.text)
      .join('\n')
      .trim();
  }
  return (parts[parts.length - 1]?.text || '').trim();
}

export async function POST(request: Request) {
  try {
    const supabase = await createServerSupabase();
    const {
      data: { user },
    } = await supabase.auth.getUser();

    if (!user) {
      return NextResponse.json(
        { error: 'Войди в аккаунт, чтобы пользоваться ИИ-тьютором.' },
        { status: 401 }
      );
    }

    const body = await request.json();
    const topicId = String(body.topicId || '');
    const problem = String(body.problem || '').trim();
    const attempt = String(body.attempt || '').trim();
    const mode = body.mode as TutorMode;

    if (!topicId || !MODE_INSTRUCTIONS[mode]) {
      return NextResponse.json(
        { error: 'Некорректные параметры запроса.' },
        { status: 400 }
      );
    }

    if (problem.length < 3 || attempt.length < 3) {
      return NextResponse.json(
        {
          error:
            'Сначала напиши условие задачи и свою самостоятельную попытку решения.',
        },
        { status: 400 }
      );
    }

    // Проверяем тему и режим СОР/СОЧ прямо в базе данных
    const { data: topic, error: topicErr } = await supabase
      .from('topics')
      .select('title, section, is_assessment, courses ( grade_number, subject )')
      .eq('id', topicId)
      .single();

    if (topicErr || !topic) {
      return NextResponse.json({ error: 'Тема не найдена.' }, { status: 404 });
    }

    if (topic.is_assessment) {
      return NextResponse.json(
        {
          error:
            'Режим академической честности («Адал азамат»): во время СОР и СОЧ ИИ-помощник отключён.',
        },
        { status: 403 }
      );
    }

    const apiKey = (process.env.GEMINI_API_KEY || '')
      .trim()
      .replace(/^["']|["']$/g, '');
    if (!apiKey) {
      return NextResponse.json(
        { error: 'На сервере не настроен GEMINI_API_KEY.' },
        { status: 500 }
      );
    }

    const rawCourse = topic.courses as
      | { grade_number?: number; subject?: string }
      | { grade_number?: number; subject?: string }[]
      | null;
    const course = Array.isArray(rawCourse) ? rawCourse[0] : rawCourse;
    const grade = course?.grade_number || 5;
    const subject = course?.subject || 'Математика';

    const chosen = MODE_INSTRUCTIONS[mode];

    const systemPrompt = `Ты — доброжелательный ИИ-тьютор по предмету «${subject}» для ученика ${grade} класса (программа Республики Казахстан).
СТРОГИЕ ПРАВИЛА:
1. НИКОГДА не выдавай готовый конечный ответ на задачу ученика! Твоя цель — научить думать самостоятельно.
2. Отвечай на русском языке, понятно для ученика ${grade} класса, кратко (4–7 предложений).
3. Поддерживай и хвали за самостоятельную попытку.
4. Для математических формул используй запись LaTeX в одинарных знаках доллара, например $\\frac{3}{4}$ или $x + 5 = 12$.
5. Текущая тема урока: «${topic.title}» (раздел: ${topic.section}).`;

    const userPrompt = `Условие задачи ученика: ${problem}
Самостоятельная попытка ученика: ${attempt}
Выбранная функция помощи: ${chosen.label}.
Инструкция: ${chosen.prompt}`;

    let geminiData: {
      candidates?: { content?: { parts?: GeminiPart[] } }[];
    } | null = null;
    let lastStatus = 502;
    let lastReason = 'Нет ответа от Gemini';
    const attempts: string[] = [];

    for (const model of MODELS) {
      const endpoint =
        'https://generativelanguage.googleapis.com/v1beta/models/' +
        encodeURIComponent(model) +
        ':generateContent';

      const res = await fetch(endpoint, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'x-goog-api-key': apiKey,
        },
        body: JSON.stringify({
          systemInstruction: {
            parts: [{ text: systemPrompt }],
          },
          contents: [
            {
              role: 'user',
              parts: [{ text: userPrompt }],
            },
          ],
          generationConfig: {
            temperature: 0.6,
            maxOutputTokens: 2048,
          },
        }),
      });

      if (res.ok) {
        geminiData = await res.json();
        break;
      }

      lastStatus = res.status;
      const errText = await res.text();
      console.error(`Tutor Gemini error (${model}, ${res.status}):`, errText);

      try {
        const errJson = JSON.parse(errText);
        lastReason = errJson?.error?.message || errText.slice(0, 180);
      } catch {
        lastReason = errText.slice(0, 180);
      }

      attempts.push(`${model}:${res.status}`);

      if (res.status === 400 && lastReason.toLowerCase().includes('api key')) {
        break;
      }

      if (res.status === 503 || res.status === 429) {
        await new Promise((r) => setTimeout(r, 400));
      }
    }

    if (!geminiData) {
      return NextResponse.json(
        {
          error: `Ошибка ИИ-тьютора (${lastStatus}) [${attempts.join(', ')}]: ${lastReason}`,
        },
        { status: 502 }
      );
    }

    const reply = extractReplyText(geminiData);
    if (!reply) {
      return NextResponse.json(
        { error: 'Модель вернула пустой ответ. Попробуй ещё раз.' },
        { status: 502 }
      );
    }

    return NextResponse.json({ reply, modeLabel: chosen.label });
  } catch (err) {
    console.error('api/tutor error:', err);
    return NextResponse.json(
      { error: 'Не удалось получить ответ от ИИ-тьютора. Попробуй через минуту.' },
      { status: 500 }
    );
  }
}

  const starsMap = new Map<

      practiceDone: boolean;
      practiceSolved: number;

import { Suspense } from 'react';
import { connection } from 'next/server';
import { notFound } from 'next/navigation';
import { createServerSupabase } from '@/lib/supabase/server';
import CourseWorkspace from '@/components/lesson/CourseWorkspace';
import type { ClassScheduleItem, Course, Profile, Topic } from '@/types/db';

async function CourseDetailContent({
  params,
  searchParams,
}: {
  params: Promise<{ slug: string }>;
  searchParams: Promise<{ topic?: string; tab?: string }>;
}) {
  await connection();
  const [{ slug }, { topic: initialTopicId, tab: initialTab }] = await Promise.all([
    params,
    searchParams,
  ]);

  const supabase = await createServerSupabase();

  const { data: course } = await supabase
    .from('courses')
    .select('*')
    .eq('slug', slug)
    .single();

  if (!course) notFound();

  const { data: topics } = await supabase
    .from('topics')
    .select('*')
    .eq('course_id', course.id)
    .order('order_index', { ascending: true });

  const {
    data: { user },
  } = await supabase.auth.getUser();

  let profile: Profile | null = null;
  let completedIds: string[] = [];
  let classrooms: { id: string; name: string; join_code: string }[] = [];
  let schedules: ClassScheduleItem[] = [];

  if (user) {
    const { data: prof } = await supabase
      .from('profiles')
      .select('*')
      .eq('id', user.id)
      .single();
    profile = prof as Profile | null;

    const { data: prog } = await supabase
      .from('topic_progress')
      .select('topic_id')
      .eq('student_id', user.id)
      .eq('completed', true);

    completedIds = (prog || []).map((p: { topic_id: string }) => p.topic_id);

    if (profile?.role === 'teacher') {
      const [{ data: clsRows }, { data: schRows }] = await Promise.all([
        supabase.from('classrooms').select('id, name, join_code').order('name'),
        supabase.from('class_schedule').select('*'),
      ]);
      classrooms = clsRows || [];
      schedules = (schRows as ClassScheduleItem[]) || [];
    } else if (profile?.classroom_id) {
      const { data: schRows } = await supabase
        .from('class_schedule')
        .select('*')
        .eq('classroom_id', profile.classroom_id);
      schedules = (schRows as ClassScheduleItem[]) || [];
    }
  }

  return (
    <CourseWorkspace
      course={course as Course}
      initialTopics={(topics as Topic[]) || []}
      initialCompletedIds={completedIds}
      profile={profile}
      classrooms={classrooms}
      initialSchedules={schedules}
      initialTopicId={initialTopicId}
      initialTab={initialTab}
    />
  );
}

export default function CourseDetailPage({
  params,
  searchParams,
}: {
  params: Promise<{ slug: string }>;
  searchParams: Promise<{ topic?: string; tab?: string }>;
}) {
  return (
    <Suspense
      fallback={
        <div className="min-h-screen flex items-center justify-center text-sm font-semibold text-slate-500">
          Загрузка программы курса...
        </div>
      }
    >
      <CourseDetailContent params={params} searchParams={searchParams} />
    </Suspense>
  );
}

    e.preventDefault();
    setError('');
    const cleanLogin = normalizeStudentLogin(studentLogin);

      setError('Введи логин с карточки и 4-значный PIN-код.');
      return;
    }

    setLoading(true);
    const { error: signErr } = await supabase.auth.signInWithPassword({
      email: studentLoginToEmail(cleanLogin),
      password: studentPinToPassword(cleanPin),
    });
    setLoading(false);

    if (signErr) {
      setError('Неверный логин или PIN-код. Проверь свою карточку.');
      return;
    }

  }

    e.preventDefault();
    setError('');

      email: email.trim().toLowerCase(),

    setLoading(false);

      return;
    }

    router.push('/today');
    router.refresh();
  }

  async function handleRegister(e: React.FormEvent) {
    e.preventDefault();
    setError('');
    setLoading(true);

    try {
      const res = await fetch('/api/auth/register', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          fullName,
          email,
          password,
          role,
          inviteCode,
          classCode,
        }),
      });
      const data = await res.json();

      if (!res.ok) {
        setError(data.error || 'Не удалось создать аккаунт.');
        setLoading(false);
        return;
      }

      const { error: signErr } = await supabase.auth.signInWithPassword({
        email: email.trim().toLowerCase(),
        password,
      });

      if (signErr) {
        setError('Аккаунт создан, но войти не удалось. Попробуй войти вручную.');
        setView('email_login');
        setLoading(false);

  const schedules: TodayScheduleCard[] = [];
  for (const row of rawSchedules) {
    const topic = topicMap.get(row.topic_id);
    if (!topic) continue;

    const tasks = topicTasksMap.get(topic.id) || [];
    let doneTasks = 0;

    let maxPoints = 0;

    for (const task of tasks) {
      maxPoints += task.points;

      doneTasks,
      earnedPoints,
      maxPoints,
      isCompleted: completedSet.has(topic.id),
    });
  }

  // Недавно проверенные учителем работы (для ученика)

            .in('id', gTopicIds)
        : { data: [] };

      const gTaskMap = new Map((gTasks || []).map((t) => [t.id, t]));
      const gTopicMap = new Map((gTopics || []).map((t) => [t.id, t]));

      for (const g of gradedRows) {
        const task = gTaskMap.get(g.task_id);
        const topic = task ? gTopicMap.get(task.topic_id) : null;
        const course = topic ? courseMap.get(topic.course_id) : null;

          topicTitle: topic.title,
          courseSlug: course.slug,
          score: Number(g.score || 0),
          maxPoints: Number(task.points || 2),
          teacherComment: g.teacher_comment || '',
        });
      }
    }
  }

  return (
    <TodayClient
      profile={profile}
      studentClassroom={studentClassroom}
      teacherClassrooms={teacherClassrooms}
      schedules={schedules}
      recentGraded={recentGraded}
      courses={courses}
      stats={{ completedTopics, simSolved, bestStreak }}
      todayIso={todayIso}
      tomorrowIso={tomorrowIso}
    />
  );

function getTopicStars(t: MapTopicItem): {
  theoryStar: boolean;
  simStar: boolean;
  hwStar: boolean;
  earned: number;
  possible: number;
} {
  const theoryStar = Boolean(t.slidesDone || t.completed);
  const simStar = Boolean(t.practiceDone || t.simSolved >= 3);
  const hasHw = t.hwTotal > 0;
  const hwStar = hasHw && (Boolean(t.homeworkDone) || t.hwDone >= t.hwTotal);
  const possible = hasHw ? 3 : 2;
  const earned = (theoryStar ? 1 : 0) + (simStar ? 1 : 0) + (hwStar ? 1 : 0);
  return { theoryStar, simStar, hwStar, earned, possible };
}

    function handleClickOutside(e: MouseEvent) {
      if (menuRef.current && !menuRef.current.contains(e.target as Node)) {
        setTeacherMenuOpen(false);
      }
    }
    document.addEventListener('mousedown', handleClickOutside);

    return () => {
      window.removeEventListener('mathhub-sound-change', syncSound);
      document.removeEventListener('mousedown', handleClickOutside);
    };
  }, []);

  function toggleSound() {
    const next = !soundOn;
    setSoundEnabled(next);
    setSoundOn(next);
    if (next) playSound('click');
  }

  async function handleLogout() {
    await supabase.auth.signOut();
    router.push('/login');
    router.refresh();
  }

  return (
    <>
      <header className="sticky top-0 z-40 bg-white/90 backdrop-blur-md border-b border-slate-200/80">

'use client';

import { useEffect, useState } from 'react';
import { playSound } from '@/lib/audio/sound';

interface BeforeInstallPromptEvent extends Event {
  prompt: () => Promise<void>;
  userChoice: Promise<{ outcome: 'accepted' | 'dismissed'; platform: string }>;
}

const DISMISS_KEY = 'mathhub_pwa_dismissed';

export default function PwaRegister() {
  const [deferredPrompt, setDeferredPrompt] =
    useState<BeforeInstallPromptEvent | null>(null);
  const [visible, setVisible] = useState(false);

  useEffect(() => {
    if ('serviceWorker' in navigator) {
      navigator.serviceWorker.register('/sw.js').catch(() => {});
    }

    const isStandalone =
      window.matchMedia('(display-mode: standalone)').matches ||
      (window.navigator as Navigator & { standalone?: boolean }).standalone === true;
    if (isStandalone) return;

    try {
      if (window.localStorage.getItem(DISMISS_KEY) === '1') return;
    } catch {
      // Игнорируем ограничения приватного режима
    }

    function handleBeforeInstall(e: Event) {
      e.preventDefault();
      setDeferredPrompt(e as BeforeInstallPromptEvent);
      setVisible(true);
    }

    function handleInstalled() {
      setVisible(false);
      setDeferredPrompt(null);
    }

    window.addEventListener('beforeinstallprompt', handleBeforeInstall);
    window.addEventListener('appinstalled', handleInstalled);
    return () => {
      window.removeEventListener('beforeinstallprompt', handleBeforeInstall);
      window.removeEventListener('appinstalled', handleInstalled);
    };
  }, []);

  async function handleInstallClick() {
    if (!deferredPrompt) return;
    playSound('click');
    await deferredPrompt.prompt();
    const choice = await deferredPrompt.userChoice;
    if (choice.outcome === 'accepted') {
      playSound('celebrate');
    }
    setDeferredPrompt(null);
    setVisible(false);
  }

  function handleDismiss() {
    playSound('click');
    setVisible(false);
    try {
      window.localStorage.setItem(DISMISS_KEY, '1');
    } catch {
      // Игнорируем ограничения приватного режима
    }
  }

  if (!visible || !deferredPrompt) return null;

  return (
    <div className="fixed bottom-20 md:bottom-5 left-4 z-40 print:hidden animate-pop">
      <div className="bg-white/95 backdrop-blur-md border-2 border-indigo-300 shadow-xl rounded-2xl px-3.5 py-2.5 flex items-center gap-2.5">
        <button
          type="button"
          onClick={handleInstallClick}
          className="inline-flex items-center gap-2 text-xs font-extrabold text-indigo-700 hover:text-indigo-900 transition"
        >
          <span className="text-base leading-none">📲</span>
          <span>Установить MathHub</span>
        </button>
        <button
          type="button"
          onClick={handleDismiss}
          title="Скрыть"
          className="w-6 h-6 rounded-lg bg-slate-100 hover:bg-slate-200 text-slate-500 text-[11px] font-bold"
        >
          ✕
        </button>
      </div>
    </div>
  );
}

export function drawBoardBackground(
  ctx: CanvasRenderingContext2D,
  width: number,
  height: number,
  bg: BoardBg
): void {
  if (bg === 'plain') {
    ctx.fillStyle = '#FFFFFF';
    ctx.fillRect(0, 0, width, height);
    return;
  }

  if (bg === 'chalk') {
    ctx.fillStyle = '#0E2F25';
    ctx.fillRect(0, 0, width, height);

    ctx.strokeStyle = 'rgba(255, 255, 255, 0.08)';
    ctx.lineWidth = 1;
    ctx.beginPath();
    for (let x = GRID_STEP; x < width; x += GRID_STEP) {
      ctx.moveTo(x + 0.5, 0);
      ctx.lineTo(x + 0.5, height);
    }
    for (let y = GRID_STEP; y < height; y += GRID_STEP) {
      ctx.moveTo(0, y + 0.5);
      ctx.lineTo(width, y + 0.5);
    }
    ctx.stroke();
    return;
  }

  // Белая бумага для 'grid' и 'coords'
  ctx.fillStyle = '#FFFFFF';
  ctx.fillRect(0, 0, width, height);

  ctx.strokeStyle = bg === 'coords' ? '#E2E8F0' : '#DBEAFE';
  ctx.lineWidth = 1;
  ctx.beginPath();
  for (let x = GRID_STEP; x < width; x += GRID_STEP) {
    ctx.moveTo(x + 0.5, 0);
    ctx.lineTo(x + 0.5, height);
  }
  for (let y = GRID_STEP; y < height; y += GRID_STEP) {
    ctx.moveTo(0, y + 0.5);
    ctx.lineTo(width, y + 0.5);
  }
  ctx.stroke();

  if (bg === 'coords') {
    const ox = Math.round(width / 2 / GRID_STEP) * GRID_STEP;
    const oy = Math.round(height / 2 / GRID_STEP) * GRID_STEP;

    ctx.strokeStyle = '#334155';
    ctx.fillStyle = '#334155';
    ctx.lineWidth = 2;

    // Ось X

    ctx.moveTo(12, oy);
    ctx.lineTo(width - 14, oy);

    ctx.beginPath();
    ctx.moveTo(width - 8, oy);
    ctx.lineTo(width - 20, oy - 6);
    ctx.lineTo(width - 20, oy + 6);
    ctx.closePath();
    ctx.fill();

    // Ось Y

    ctx.moveTo(ox, height - 12);
    ctx.lineTo(ox, 14);
    ctx.stroke();
    ctx.beginPath();
    ctx.moveTo(ox, 8);
    ctx.lineTo(ox - 6, 20);
    ctx.lineTo(ox + 6, 20);
    ctx.closePath();
    ctx.fill();

    // Штрихи и подписи осей

    ctx.textAlign = 'center';
    ctx.textBaseline = 'top';

    const maxUnitsX = Math.floor(Math.max(ox, width - ox) / GRID_STEP);
    for (let u = -maxUnitsX; u <= maxUnitsX; u++) {
      if (u === 0) continue;
      const x = ox + u * GRID_STEP;
      if (x < 24 || x > width - 28) continue;
      ctx.beginPath();
      ctx.moveTo(x, oy - 4);
      ctx.lineTo(x, oy + 4);
      ctx.stroke();

    for (let u = -maxUnitsY; u <= maxUnitsY; u++) {

export function drawBoardElement(
  ctx: CanvasRenderingContext2D,
  el: BoardElement
): void {
  ctx.save();
  ctx.lineCap = 'round';
  ctx.lineJoin = 'round';

 if ('points' in el) {
    const pts = el.points;
    if (pts.length === 0) {
      ctx.restore();
      return;
    }

    const isHighlighter = el.kind === 'highlighter';
    ctx.strokeStyle = el.color;
    ctx.fillStyle = el.color;
    ctx.lineWidth = isHighlighter ? el.width * 3.2 : el.width;
    ctx.globalAlpha = isHighlighter ? 0.34 : 1;

    if (pts.length === 1) {
      ctx.beginPath();
      ctx.arc(pts[0].x, pts[0].y, ctx.lineWidth / 2, 0, Math.PI * 2);
      ctx.fill();
      ctx.restore();
      return;
    }

    ctx.beginPath();
    ctx.moveTo(pts[0].x, pts[0].y);
    for (let i = 1; i < pts.length - 1; i++) {
      const midX = (pts[i].x + pts[i + 1].x) / 2;
      const midY = (pts[i].y + pts[i + 1].y) / 2;
      ctx.quadraticCurveTo(pts[i].x, pts[i].y, midX, midY);
    }
    const last = pts[pts.length - 1];
    ctx.lineTo(last.x, last.y);
    ctx.stroke();
    ctx.restore();
    return;
  }

  ctx.strokeStyle = el.color;
  ctx.fillStyle = el.color;
  ctx.lineWidth = el.width;

  const { start, end } = el;

  if (el.kind === 'line') {
    ctx.beginPath();
    ctx.moveTo(start.x, start.y);
    ctx.lineTo(end.x, end.y);
    ctx.stroke();
  } else if (el.kind === 'arrow') {
    const dx = end.x - start.x;
    const dy = end.y - start.y;
    const angle = Math.atan2(dy, dx);
    const headLen = Math.max(12, el.width * 3.2);

    ctx.beginPath();
    ctx.moveTo(start.x, start.y);
    ctx.lineTo(end.x, end.y);
    ctx.stroke();

    if (Math.hypot(dx, dy) > 4) {
      ctx.beginPath();
      ctx.moveTo(end.x, end.y);
      ctx.lineTo(
        end.x - headLen * Math.cos(angle - Math.PI / 6),
        end.y - headLen * Math.sin(angle - Math.PI / 6)
      );
      ctx.lineTo(
        end.x - headLen * Math.cos(angle + Math.PI / 6),
        end.y - headLen * Math.sin(angle + Math.PI / 6)
      );
      ctx.closePath();
      ctx.fill();
    }
  } else if (el.kind === 'rect') {
    const x = Math.min(start.x, end.x);
    const y = Math.min(start.y, end.y);
    const w = Math.abs(end.x - start.x);
    const h = Math.abs(end.y - start.y);
    ctx.strokeRect(x, y, w, h);
  } else if (el.kind === 'circle') {
    const r = Math.hypot(end.x - start.x, end.y - start.y);
    ctx.beginPath();
    ctx.arc(start.x, start.y, r, 0, Math.PI * 2);
    ctx.stroke();
    // Точка центра окружности — удобно для задач по геометрии
    if (r > 14) {
      ctx.beginPath();
      ctx.arc(start.x, start.y, Math.max(2.5, el.width * 0.65), 0, Math.PI * 2);
      ctx.fill();
    }
  }

  ctx.restore();
}

function distToSegment(p: Point, a: Point, b: Point): number {
  const dx = b.x - a.x;
  const dy = b.y - a.y;
  const lenSq = dx * dx + dy * dy;
  if (lenSq === 0) return Math.hypot(p.x - a.x, p.y - a.y);
  let t = ((p.x - a.x) * dx + (p.y - a.y) * dy) / lenSq;
  t = Math.max(0, Math.min(1, t));
  const projX = a.x + t * dx;
  const projY = a.y + t * dy;
  return Math.hypot(p.x - projX, p.y - projY);
}

export function hitTestElement(
  el: BoardElement,
  p: Point,
  baseRadius = 14
): boolean {
  const threshold = Math.max(baseRadius, el.width * 1.6);

  if ('points' in el) {
    const pts = el.points;
    if (pts.length === 0) return false;
    if (pts.length === 1) {
      return Math.hypot(p.x - pts[0].x, p.y - pts[0].y) <= threshold;
    }
    const r = el.kind === 'highlighter' ? Math.max(threshold, el.width * 2.2) : threshold;
    for (let i = 0; i < pts.length - 1; i++) {
      if (distToSegment(p, pts[i], pts[i + 1]) <= r) return true;
    }
    return false;
  }

  const { start, end } = el;
  if (el.kind === 'line' || el.kind === 'arrow') {
    return distToSegment(p, start, end) <= threshold;
  }

  if (el.kind === 'rect') {
    const x1 = Math.min(start.x, end.x);
    const x2 = Math.max(start.x, end.x);
    const y1 = Math.min(start.y, end.y);
    const y2 = Math.max(start.y, end.y);
    const tl = { x: x1, y: y1 };
    const tr = { x: x2, y: y1 };
    const br = { x: x2, y: y2 };
    const bl = { x: x1, y: y2 };
    return (
      distToSegment(p, tl, tr) <= threshold ||
      distToSegment(p, tr, br) <= threshold ||
      distToSegment(p, br, bl) <= threshold ||
      distToSegment(p, bl, tl) <= threshold
    );
  }

  if (el.kind === 'circle') {
    const r = Math.hypot(end.x - start.x, end.y - start.y);
    const d = Math.hypot(p.x - start.x, p.y - start.y);
    return Math.abs(d - r) <= threshold || d <= Math.min(12, threshold);
  }

  return false;
}

            <div className="flex items-center gap-1 p-1 rounded-xl bg-slate-900/85 text-white backdrop-blur text-[11px] font-bold shadow-md">
              {BG_OPTIONS.map((b) => (
                <button
                  key={b.id}
                  type="button"
                  onClick={() => setPageBg(b.id)}
                  title={b.label}
                  className={`px-2 py-1 rounded-lg transition ${
                    currentPage?.bg === b.id
                      ? 'bg-amber-400 text-slate-950'
                      : 'text-slate-300 hover:bg-slate-800'
                  }`}
                >
                  <span className="mr-1">{b.icon}</span>
                  <span className="hidden sm:inline">{b.label}</span>
                </button>
              ))}
            </div>
          </div>
        )}

        {toast && (
          <div className="absolute top-3 right-3 z-30 px-3.5 py-2 rounded-xl bg-emerald-600 text-white text-xs font-extrabold shadow-lg animate-pop">
            {toast}
          </div>
        )}

        {/* Шторка-занавес (как в mozaBook / SMART Notebook) */}
        {shadeOpen && (
          <div
            style={{ top: `${shadeTopPct}%` }}
            className="absolute inset-x-0 bottom-0 z-20 bg-slate-900/95 backdrop-blur-md border-t-4 border-amber-400 shadow-2xl pointer-events-auto flex flex-col"
          >
            <div
              onPointerDown={handleShadePointerDown}
              onPointerMove={handleShadePointerMove}
              className="h-10 cursor-ns-resize touch-none flex items-center justify-between px-4 bg-slate-800/90 text-amber-300 text-xs font-extrabold select-none"
            >
              <span>🪟 Шторка (потяни вверх или вниз, чтобы открыть шаг)</span>
              <button
                type="button"
                onClick={(e) => {
                  e.stopPropagation();
                  playSound('click');
                  setShadeOpen(false);
                }}
                className="px-2.5 py-1 rounded-lg bg-slate-700 hover:bg-rose-600 text-white text-[11px]"
              >
                Убрать ✕
              </button>
            </div>
            <div className="flex-1 flex items-center justify-center text-slate-500 text-xs font-bold select-none pointer-events-none">
              Скрытая область экрана
            </div>
          </div>
        )}
      </div>

      {/* Плавающий пульт MathBoard внизу экрана */}
      <div className="fixed bottom-3 left-1/2 -translate-x-1/2 z-[60] max-w-[97vw] print:hidden">
        {toolbarMinimized ? (
          <div className="bg-slate-900/95 text-white backdrop-blur rounded-2xl shadow-2xl border border-slate-700 px-3 py-2 flex items-center gap-2">
            <button
              type="button"
              onClick={() => setToolbarMinimized(false)}
              className="px-3 py-1.5 rounded-xl bg-indigo-500 hover:bg-indigo-600 text-xs font-extrabold"
            >
              🎨 Развернуть пульт доски
            </button>
            <button
              type="button"
              onClick={onExitBoard}
              className="px-3 py-1.5 rounded-xl bg-rose-500 hover:bg-rose-600 text-xs font-extrabold"
            >
              ✕ Выйти
            </button>
          </div>
        ) : (
          <div className="bg-slate-900/95 text-white backdrop-blur-md rounded-2xl shadow-2xl border border-slate-700 px-2.5 py-2 space-y-2">
            {/* Верхний ряд: Режим экрана + Инструменты рисования */}
            <div className="flex flex-wrap items-center justify-center gap-1.5">
              {/* Режим: Стекло / Сплит / Доска */}
              <div className="flex items-center gap-0.5 bg-slate-800 p-0.5 rounded-xl">
                {VIEW_MODES.map((m) => (
                  <button
                    key={m.id}
                    type="button"
                    onClick={() => switchViewMode(m.id)}
                    className={`px-2.5 py-1.5 rounded-lg text-[11px] font-extrabold transition ${
                      viewMode === m.id
                        ? 'bg-emerald-500 text-slate-950 shadow-xs'
                        : 'text-slate-300 hover:text-white'
                    }`}
                  >
                    {m.icon} <span className="hidden sm:inline">{m.label}</span>
                  </button>
                ))}
              </div>

              <div className="h-5 w-px bg-slate-700 hidden sm:block" />

              {/* Инструменты */}
              <div className="flex flex-wrap items-center justify-center gap-1">
                {TOOLS.filter((t) => !t.glassOnly || viewMode === 'glass').map((t) => {
                  const active = tool === t.id;
                  return (
                    <button
                      key={t.id}
                      type="button"
                      title={t.label}
                      onClick={() => {
                        playSound('click');
                        setTool(t.id);
                      }}
                      className={`h-8 min-w-[34px] px-2 rounded-xl text-xs font-extrabold transition flex items-center justify-center ${
                        active
                          ? 'bg-amber-400 text-slate-950 shadow-xs scale-105'
                          : 'bg-slate-800 hover:bg-slate-700 text-slate-200'
                      }`}
                    >
                      {t.icon}
                    </button>
                  );
                })}
              </div>
            </div>

            {/* Нижний ряд: Цвета, толщина, сетка, шторка, история, сохранение */}
            <div className="flex flex-wrap items-center justify-center gap-1.5 pt-1 border-t border-slate-800">
              {/* Палитра цветов */}
              <div className="flex items-center gap-1">
                {BOARD_COLORS.map((c) => (
                  <button
                    key={c.value}
                    type="button"
                    title={c.label}
                    onClick={() => {
                      setColor(c.value);
                      if (tool === 'cursor' || tool === 'eraser' || tool === 'laser') {
                        setTool('pen');
                      }
                    }}
                    className={`w-6 h-6 rounded-full border-2 transition ${
                      color === c.value
                        ? 'border-amber-400 scale-115'
                        : 'border-slate-500 hover:scale-105'
                    }`}
                    style={{ backgroundColor: c.value }}
                  />
                ))}
              </div>

              {/* Толщина */}
              <div className="flex items-center gap-0.5 bg-slate-800 p-0.5 rounded-xl">
                {BOARD_WIDTHS.map((w) => (
                  <button
                    key={w.value}
                    type="button"
                    onClick={() => setWidth(w.value)}
                    className={`px-2 py-1 rounded-lg text-[11px] font-extrabold transition ${
                      width === w.value
                        ? 'bg-indigo-500 text-white'
                        : 'text-slate-300 hover:text-white'
                    }`}
                  >
                    {w.label}
                  </button>
                ))}
              </div>

              {/* Прилипание фигур к сетке */}
              {viewMode !== 'glass' && (
                <button
                  type="button"
                  title="Прилипание фигур к узлам сетки"
                  onClick={() => {
                    playSound('click');
                    setSnapToGrid((v) => !v);
                  }}
                  className={`px-2 py-1 rounded-xl text-[11px] font-bold transition ${
                    snapToGrid
                      ? 'bg-indigo-500/30 text-indigo-200 border border-indigo-400/50'
                      : 'bg-slate-800 text-slate-400'
                  }`}
                >
                  🧲 Сетка
                </button>
              )}

              {/* Шторка */}
              <button
                type="button"
                title="Шторка для постепенного открытия решения"
                onClick={() => {
                  playSound('click');
                  setShadeOpen((v) => !v);
                }}
                className={`px-2 py-1 rounded-xl text-[11px] font-bold transition ${
                  shadeOpen
                    ? 'bg-amber-400 text-slate-950'
                    : 'bg-slate-800 hover:bg-slate-700 text-slate-200'
                }`}
              >
                🪟 Шторка
              </button>

              <div className="h-5 w-px bg-slate-700 hidden sm:block" />

              {/* Undo / Redo / Clear */}
              <div className="flex items-center gap-1">
                <button
                  type="button"
                  disabled={!canUndo}
                  onClick={handleUndo}
                  title="Отменить (Ctrl+Z)"
                  className="px-2 py-1 rounded-xl bg-slate-800 hover:bg-slate-700 disabled:opacity-35 text-xs font-bold"
                >
                  ↩
                </button>
                <button
                  type="button"
                  disabled={!canRedo}
                  onClick={handleRedo}
                  title="Вернуть (Ctrl+Y)"
                  className="px-2 py-1 rounded-xl bg-slate-800 hover:bg-slate-700 disabled:opacity-35 text-xs font-bold"
                >
                  ↪
                </button>
                <button
                  type="button"
                  disabled={!hasElements}
                  onClick={handleClearPage}
                  title="Очистить лист"
                  className="px-2 py-1 rounded-xl bg-slate-800 hover:bg-rose-600 disabled:opacity-35 text-xs font-bold"
                >
                  🗑️
                </button>
              </div>

              {/* Сохранение */}
              <div className="flex items-center gap-1">
                <button
                  type="button"
                  onClick={handleDownloadPng}
                  title="Скачать лист доски как PNG"
                  className="px-2.5 py-1 rounded-xl bg-slate-800 hover:bg-slate-700 text-xs font-bold text-slate-200"
                >
                  ⬇ PNG
                </button>

                {isTeacher && topic && (
                  <button
                    type="button"
                    disabled={savingToLesson}
                    onClick={handleSaveToLesson}
                    title="Сохранить этот лист в материалы урока для учеников"
                    className="px-2.5 py-1 rounded-xl bg-emerald-600 hover:bg-emerald-500 disabled:opacity-50 text-white text-xs font-extrabold transition"
                  >
                    {savingToLesson ? '...' : '💾 В урок'}
                  </button>
                )}
              </div>

              <button
                type="button"
                onClick={() => setToolbarMinimized(true)}
                title="Свернуть пульт"
                className="px-2 py-1 rounded-xl bg-slate-800 hover:bg-slate-700 text-slate-400 text-xs font-bold"
              >
                ⌄
              </button>

              <button
                type="button"
                onClick={onExitBoard}
                className="px-2.5 py-1 rounded-xl bg-rose-500 hover:bg-rose-600 text-white text-xs font-extrabold"
              >
                ✕ Выйти
              </button>
            </div>
          </div>
        )}
      </div>
    </>
  );
}
