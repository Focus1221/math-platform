'use client';

import { useMemo, useState } from 'react';
import Link from 'next/link';

export interface ClassroomRow {
  id: string;
  name: string;
  join_code: string;
  grade_number: number;
}

export interface StudentRow {
  id: string;
  full_name: string;
  email: string;
  classroom_id: string | null;
  login_username?: string | null;
  pin_code?: string | null;
}

export interface JournalCourse {
  id: string;
  title: string;
  grade_number: number;
}

export interface JournalTopic {
  id: string;
  course_id: string;
  order_index: number;
  quarter: number;
  title: string;
  is_assessment: boolean;
  maxPoints: number;
}

export interface ProgressEntry {
  student_id: string;
  topic_id: string;
  completed: boolean;
  sim_correct_count: number;
}

export interface TopicScoreEntry {
  student_id: string;
  topic_id: string;
  score: number;
}

interface TopicHeatStats {
  topic: JournalTopic;
  submittedCount: number;
  completedCount: number;
  avgPct: number | null;
  lowStudents: { id: string; name: string; score: number }[];
  level: 'none' | 'good' | 'warn' | 'gap';
}

interface Props {
  initialClassrooms: ClassroomRow[];
  initialStudents: StudentRow[];
  courses: JournalCourse[];
  topics: JournalTopic[];
  progressList: ProgressEntry[];
  topicScores: TopicScoreEntry[];
}

function getScoreBadgeClasses(score: number, maxPoints: number, heatmapOn: boolean): string {
  if (!heatmapOn || maxPoints <= 0) {
    return 'bg-indigo-100 text-indigo-800';
  }
  const pct = (score / maxPoints) * 100;
  if (pct >= 80) return 'bg-emerald-100 text-emerald-800 border border-emerald-300';
  if (pct >= 50) return 'bg-amber-100 text-amber-800 border border-amber-300';
  return 'bg-rose-100 text-rose-800 border border-rose-300';
}

export default function JournalClient({
  initialClassrooms,
  initialStudents,
  courses,
  topics,
  progressList,
  topicScores,
}: Props) {
  const [tab, setTab] = useState<'matrix' | 'classes'>('matrix');
  const [classrooms, setClassrooms] = useState<ClassroomRow[]>(initialClassrooms);
  const [students, setStudents] = useState<StudentRow[]>(initialStudents);

  // Фильтры матрицы и тепловой карты
  const [selectedClassId, setSelectedClassId] = useState<string>('all');
  const [selectedCourseId, setSelectedCourseId] = useState<string>(courses[0]?.id || '');
  const [selectedQuarter, setSelectedQuarter] = useState<number>(1);
  const [heatmapOn, setHeatmapOn] = useState<boolean>(true);
  const [focusedTopicId, setFocusedTopicId] = useState<string | null>(null);

  // Форма создания класса
  const [newClassName, setNewClassName] = useState('');
  const [newClassGrade, setNewClassGrade] = useState(5);
  const [newClassCode, setNewClassCode] = useState('');
  const [creatingClass, setCreatingClass] = useState(false);

  // Массовое добавление учеников (логин + PIN)
  const [bulkClassId, setBulkClassId] = useState<string>(initialClassrooms[0]?.id || '');
  const [bulkNamesText, setBulkNamesText] = useState('');
  const [bulkCreating, setBulkCreating] = useState(false);
  const [bulkNotice, setBulkNotice] = useState('');

  // Уведомление о сброшенном пароле / PIN
  const [resetResult, setResetResult] = useState<{
    name: string;
    subtitle: string;
    password: string;
    isPin: boolean;
  } | null>(null);
  const [busyStudentId, setBusyStudentId] = useState<string | null>(null);

  const activeBulkClassId = useMemo(() => {
    if (bulkClassId && classrooms.some((c) => c.id === bulkClassId)) {
      return bulkClassId;
    }
    return classrooms[0]?.id || '';
  }, [bulkClassId, classrooms]);

  // Быстрые словари для мгновенного рендера таблицы
  const progressMap = useMemo(() => {
    const map = new Map<string, ProgressEntry>();
    for (const p of progressList) {
      map.set(`${p.student_id}:${p.topic_id}`, p);
    }
    return map;
  }, [progressList]);

  const { scoreMap, submittedSet } = useMemo(() => {
    const scores = new Map<string, number>();
    const submitted = new Set<string>();
    for (const s of topicScores) {
      const k = `${s.student_id}:${s.topic_id}`;
      submitted.add(k);
      scores.set(k, (scores.get(k) || 0) + s.score);
    }
    return { scoreMap: scores, submittedSet: submitted };
  }, [topicScores]);

  const classMap = useMemo(
    () => new Map(classrooms.map((c) => [c.id, c])),
    [classrooms]
  );

  const visibleTopics = useMemo(
    () =>
      topics
        .filter((t) => t.course_id === selectedCourseId && t.quarter === selectedQuarter)
        .sort((a, b) => a.order_index - b.order_index),
    [topics, selectedCourseId, selectedQuarter]
  );

  const visibleStudents = useMemo(() => {
    return students
      .filter((s) => {
        if (selectedClassId === 'all') return true;
        if (selectedClassId === 'unassigned') return !s.classroom_id;
        return s.classroom_id === selectedClassId;
      })
      .sort((a, b) => a.full_name.localeCompare(b.full_name, 'ru'));
  }, [students, selectedClassId]);

  // Аналитика пробелов по каждой теме текущей четверти и выборки учеников
  const topicHeatMap = useMemo(() => {
    const statsMap = new Map<string, TopicHeatStats>();

    for (const topic of visibleTopics) {
      let submittedCount = 0;
      let completedCount = 0;
      let scoreSum = 0;
      const lowStudents: { id: string; name: string; score: number }[] = [];

      for (const st of visibleStudents) {
        const key = `${st.id}:${topic.id}`;
        const p = progressMap.get(key);
        if (p?.completed) completedCount++;

        if (topic.maxPoints > 0 && submittedSet.has(key)) {
          const sc = scoreMap.get(key) || 0;
          submittedCount++;
          scoreSum += sc;
          if ((sc / topic.maxPoints) * 100 < 60) {
            lowStudents.push({ id: st.id, name: st.full_name, score: sc });
          }
        }
      }

      let avgPct: number | null = null;
      let level: TopicHeatStats['level'] = 'none';

      if (topic.maxPoints > 0 && submittedCount > 0) {
        avgPct = Math.round((scoreSum / (submittedCount * topic.maxPoints)) * 100);
        const lowShare = lowStudents.length / submittedCount;
        if (avgPct < 55 || lowShare >= 0.5) {
          level = 'gap';
        } else if (avgPct < 75 || lowShare >= 0.3) {
          level = 'warn';
        } else {
          level = 'good';
        }
      }

      statsMap.set(topic.id, {
        topic,
        submittedCount,
        completedCount,
        avgPct,
        lowStudents,
        level,
      });
    }

    return statsMap;
  }, [visibleTopics, visibleStudents, progressMap, scoreMap, submittedSet]);

  const gapTopics = useMemo(() => {
    return Array.from(topicHeatMap.values())
      .filter((item) => item.level === 'gap' || item.level === 'warn')
      .sort((a, b) => (a.avgPct ?? 100) - (b.avgPct ?? 100));
  }, [topicHeatMap]);

  const focusedStats = focusedTopicId ? topicHeatMap.get(focusedTopicId) || null : null;

  async function handleCreateClass(e: React.FormEvent) {
    e.preventDefault();
    if (!newClassName.trim()) return;
    setCreatingClass(true);
    try {
      const res = await fetch('/api/teacher/classes', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          name: newClassName.trim(),
          gradeNumber: newClassGrade,
          joinCode: newClassCode.trim(),
        }),
      });
      const data = await res.json();
      if (!res.ok) {
        alert(data.error || 'Не удалось создать класс');
        return;
      }
      const created = data.classroom as ClassroomRow;
      setClassrooms((prev) => [...prev, created]);
      if (!bulkClassId) setBulkClassId(created.id);
      setNewClassName('');
      setNewClassCode('');
    } finally {
      setCreatingClass(false);
    }
  }

  async function handleDeleteClass(cls: ClassroomRow) {
    if (
      !confirm(
        `Удалить класс ${cls.name}? Ученики не удалятся, но перейдут в «Без класса».`
      )
    ) {
      return;
    }
    const res = await fetch(`/api/teacher/classes?id=${cls.id}`, {
      method: 'DELETE',
    });
    if (res.ok) {
      setClassrooms((prev) => prev.filter((c) => c.id !== cls.id));
      setStudents((prev) =>
        prev.map((s) => (s.classroom_id === cls.id ? { ...s, classroom_id: null } : s))
      );
      if (selectedClassId === cls.id) setSelectedClassId('all');
    }
  }

  async function handleBulkAddStudents(e: React.FormEvent) {
    e.preventDefault();
    if (!activeBulkClassId || !bulkNamesText.trim()) return;
    setBulkCreating(true);
    setBulkNotice('');
    try {
      const res = await fetch('/api/teacher/bulk-students', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          classroomId: activeBulkClassId,
          namesText: bulkNamesText,
        }),
      });
      const data = await res.json();
      if (!res.ok) {
        alert(data.error || 'Не удалось добавить учеников');
        return;
      }
      const added = (data.created || []) as StudentRow[];
      if (added.length > 0) {
        setStudents((prev) => [...prev, ...added]);
        setBulkNamesText('');
        setBulkNotice(
          `Создано учеников: ${added.length}. Теперь можно сразу нажать «🖨️ Карточки A4» у выбранного класса!`
        );
      }
      if (Array.isArray(data.failed) && data.failed.length > 0) {
        alert(
          `Не удалось создать (${data.failed.length}): ` +
            data.failed
              .map((f: { name: string; reason: string }) => `${f.name} (${f.reason})`)
              .join(', ')
        );
      }
    } finally {
      setBulkCreating(false);
    }
  }

  async function handleAssignClass(studentId: string, classroomId: string) {
    setBusyStudentId(studentId);
    try {
      const nextId = classroomId || null;
      const res = await fetch('/api/teacher/classes', {
        method: 'PATCH',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ studentId, classroomId: nextId }),
      });
      if (res.ok) {
        setStudents((prev) =>
          prev.map((s) => (s.id === studentId ? { ...s, classroom_id: nextId } : s))
        );
      }
    } finally {
      setBusyStudentId(null);
    }
  }

  async function handleResetPassword(student: StudentRow) {
    const label = student.login_username ? 'PIN-код' : 'пароль';
    if (!confirm(`Сбросить ${label} ученику ${student.full_name}?`)) return;
    setBusyStudentId(student.id);
    try {
      const res = await fetch('/api/teacher/reset-password', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ studentId: student.id }),
      });
      const data = await res.json();
      if (!res.ok) {
        alert(data.error || 'Ошибка сброса пароля');
        return;
      }

      const isPin = Boolean(data.isPin);
      if (isPin) {
        setStudents((prev) =>
          prev.map((s) =>
            s.id === student.id ? { ...s, pin_code: data.tempPassword } : s
          )
        );
      }

      setResetResult({
        name: student.full_name,
        subtitle: isPin
          ? `логин: ${data.loginUsername || student.login_username}`
          : student.email,
        password: data.tempPassword,
        isPin,
      });
    } finally {
      setBusyStudentId(null);
    }
  }

  async function handleRenameStudent(student: StudentRow) {
    const nextName = prompt('Новое имя и фамилия ученика:', student.full_name);
    if (!nextName || !nextName.trim() || nextName.trim() === student.full_name) {
      return;
    }

    setBusyStudentId(student.id);
    try {
      const res = await fetch('/api/teacher/students', {
        method: 'PATCH',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ studentId: student.id, fullName: nextName.trim() }),
      });
      const data = await res.json();
      if (!res.ok) {
        alert(data.error || 'Не удалось переименовать ученика');
        return;
      }
      setStudents((prev) =>
        prev.map((s) =>
          s.id === student.id ? { ...s, full_name: data.fullName } : s
        )
      );
    } finally {
      setBusyStudentId(null);
    }
  }

  async function handleDeleteStudent(student: StudentRow) {
    if (
      !confirm(
        `Удалить аккаунт ученика «${student.full_name}» вместе со всеми его баллами и работами?`
      )
    ) {
      return;
    }

    setBusyStudentId(student.id);
    try {
      const res = await fetch(`/api/teacher/students?id=${student.id}`, {
        method: 'DELETE',
      });
      const data = await res.json();
      if (!res.ok) {
        alert(data.error || 'Не удалось удалить ученика');
        return;
      }
      setStudents((prev) => prev.filter((s) => s.id !== student.id));
    } finally {
      setBusyStudentId(null);
    }
  }

  function exportCsv() {
    const esc = (v: string | number) => `"${String(v).replace(/"/g, '""')}"`;
    const headers = [
      'Ученик',
      'Класс',
      'Пройдено тем',
      'Баллы ДЗ',
      'Решено в тренажёрах',
      ...visibleTopics.map((t) => `№${t.order_index} ${t.title}`),
    ];

    const rows = visibleStudents.map((st) => {
      let doneCount = 0;
      let totalScore = 0;
      let totalSim = 0;

      const cells = visibleTopics.map((t) => {
        const key = `${st.id}:${t.id}`;
        const p = progressMap.get(key);
        const sc = scoreMap.get(key) || 0;
        const hasSub = submittedSet.has(key);
        if (p?.completed) doneCount++;
        totalScore += sc;
        totalSim += p?.sim_correct_count || 0;

        if (t.maxPoints > 0 && hasSub) return `${sc}/${t.maxPoints}`;
        if (p?.completed) return '✓';
        return '';
      });

      const clsName = st.classroom_id
        ? classMap.get(st.classroom_id)?.name || ''
        : 'Без класса';
      return [st.full_name, clsName, doneCount, totalScore, totalSim, ...cells]
        .map(esc)
        .join(';');
    });

    const csv = '\uFEFF' + [headers.map(esc).join(';'), ...rows].join('\n');
    const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' });
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = `journal-q${selectedQuarter}.csv`;
    a.click();
    URL.revokeObjectURL(url);
  }

  return (
    <div className="min-h-screen bg-[#F6F7FB]">
      <header className="sticky top-0 z-30 bg-white/95 backdrop-blur border-b border-slate-200">
        <div className="max-w-[1600px] mx-auto px-4 h-16 flex items-center justify-between gap-3">
          <div>
            <h1 className="text-base sm:text-lg font-extrabold text-slate-900">
              📊 Классы и журнал успеваемости
            </h1>
            <p className="text-xs text-slate-500">
              Классов: <strong>{classrooms.length}</strong> · Учеников:{' '}
              <strong>{students.length}</strong>
            </p>
          </div>
          <div className="flex items-center gap-2">
            <Link
              href="/teacher/submissions"
              className="px-3 py-2 rounded-xl bg-amber-50 hover:bg-amber-100 text-amber-800 border border-amber-200 text-xs font-bold transition"
            >
              📥 Проверка ДЗ
            </Link>
            <Link
              href="/today"
              className="px-3.5 py-2 rounded-xl bg-slate-100 hover:bg-slate-200 text-xs font-bold text-slate-700 transition"
            >
              ← Сегодня
            </Link>
          </div>
        </div>
      </header>

      <main className="max-w-[1600px] mx-auto px-4 py-6 space-y-5">
        {/* Переключатель разделов */}
        <div className="flex flex-wrap items-center justify-between gap-3">
          <div className="inline-flex p-1 bg-white rounded-xl border border-slate-200">
            <button
              type="button"
              onClick={() => setTab('matrix')}
              className={`px-4 py-2 rounded-lg text-xs font-bold transition ${
                tab === 'matrix' ? 'bg-indigo-600 text-white' : 'text-slate-600'
              }`}
            >
              📊 Сводный журнал
            </button>
            <button
              type="button"
              onClick={() => setTab('classes')}
              className={`px-4 py-2 rounded-lg text-xs font-bold transition ${
                tab === 'classes' ? 'bg-indigo-600 text-white' : 'text-slate-600'
              }`}
            >
              🏫 Классы и ученики ({students.length})
            </button>
          </div>

          {tab === 'matrix' && (
            <button
              type="button"
              onClick={exportCsv}
              className="btn-tactile px-4 py-2 rounded-xl bg-emerald-600 hover:bg-emerald-700 border-emerald-800 text-white text-xs font-bold shadow-xs transition"
            >
              ⬇ Скачать таблицу (Excel / CSV)
            </button>
          )}
        </div>

        {/* Баннер с новым паролем или PIN */}
        {resetResult && (
          <div className="p-4 rounded-2xl bg-emerald-50 border-2 border-emerald-300 flex flex-wrap items-center justify-between gap-3 animate-pop">
            <div className="text-xs sm:text-sm text-emerald-950 space-y-0.5">
              <div className="font-extrabold">
                🔑 {resetResult.isPin ? 'PIN-код обновлён' : 'Пароль обновлён'} для:{' '}
                {resetResult.name} ({resetResult.subtitle})
              </div>
              <div>
                {resetResult.isPin ? 'Новый PIN-код:' : 'Временный пароль:'}{' '}
                <code className="px-2 py-0.5 rounded-lg bg-white border border-emerald-300 font-mono text-base font-extrabold text-emerald-800">
                  {resetResult.password}
                </code>
              </div>
            </div>
            <div className="flex items-center gap-2">
              <button
                type="button"
                onClick={() => navigator.clipboard.writeText(resetResult.password)}
                className="px-3 py-1.5 rounded-xl bg-emerald-600 text-white text-xs font-bold"
              >
                📋 Копировать
              </button>
              <button
                type="button"
                onClick={() => setResetResult(null)}
                className="px-3 py-1.5 rounded-xl bg-white border border-emerald-200 text-xs font-bold text-slate-600"
              >
                Закрыть
              </button>
            </div>
          </div>
        )}

        {/* ВКЛАДКА 1: Сводная матрица + Тепловая карта пробелов */}
        {tab === 'matrix' && (
          <div className="space-y-4">
            <div className="bg-white rounded-2xl border border-slate-200 p-4 flex flex-wrap items-center justify-between gap-3">
              <div className="flex flex-wrap items-center gap-3">
                <select
                  value={selectedClassId}
                  onChange={(e) => {
                    setSelectedClassId(e.target.value);
                    setFocusedTopicId(null);
                  }}
                  className="px-3 py-2 text-xs font-bold bg-slate-50 border border-slate-200 rounded-xl"
                >
                  <option value="all">Все классы ({students.length})</option>
                  {classrooms.map((c) => (
                    <option key={c.id} value={c.id}>
                      Класс {c.name} ({c.join_code})
                    </option>
                  ))}
                  <option value="unassigned">Без класса</option>
                </select>

                <select
                  value={selectedCourseId}
                  onChange={(e) => {
                    setSelectedCourseId(e.target.value);
                    setFocusedTopicId(null);
                  }}
                  className="px-3 py-2 text-xs font-bold bg-slate-50 border border-slate-200 rounded-xl"
                >
                  {courses.map((c) => (
                    <option key={c.id} value={c.id}>
                      {c.title}
                    </option>
                  ))}
                </select>

                <div className="inline-flex p-1 bg-slate-100 rounded-xl">
                  {[1, 2, 3, 4].map((q) => (
                    <button
                      key={q}
                      type="button"
                      onClick={() => {
                        setSelectedQuarter(q);
                        setFocusedTopicId(null);
                      }}
                      className={`px-3 py-1.5 rounded-lg text-xs font-bold transition ${
                        selectedQuarter === q
                          ? 'bg-indigo-600 text-white'
                          : 'text-slate-600 hover:text-slate-900'
                      }`}
                    >
                      {q} четверть
                    </button>
                  ))}
                </div>
              </div>

              <div className="flex flex-wrap items-center gap-2.5">
                {heatmapOn && (
                  <div className="hidden xl:flex items-center gap-2 text-[11px] font-bold text-slate-500">
                    <span className="inline-flex items-center gap-1">
                      <span className="w-2.5 h-2.5 rounded-sm bg-emerald-400" />
                      ≥80%
                    </span>
                    <span className="inline-flex items-center gap-1">
                      <span className="w-2.5 h-2.5 rounded-sm bg-amber-400" />
                      50–79%
                    </span>
                    <span className="inline-flex items-center gap-1">
                      <span className="w-2.5 h-2.5 rounded-sm bg-rose-400" />
                      &lt;50% пробел
                    </span>
                  </div>
                )}

                <button
                  type="button"
                  onClick={() => {
                    setHeatmapOn((v) => !v);
                    setFocusedTopicId(null);
                  }}
                  className={`px-3 py-2 rounded-xl border text-xs font-extrabold transition ${
                    heatmapOn
                      ? 'bg-rose-50 border-rose-300 text-rose-800'
                      : 'bg-slate-50 border-slate-200 text-slate-600 hover:border-indigo-300'
                  }`}
                >
                  {heatmapOn ? '🔥 Тепловая карта: ВКЛ' : '🔥 Тепловая карта: ВЫКЛ'}
                </button>
              </div>
            </div>

            {/* Сводка проблемных тем для повторения перед СОР */}
            {heatmapOn && visibleStudents.length > 0 && gapTopics.length > 0 && (
              <div className="bg-white rounded-3xl border-2 border-amber-200 p-4 sm:p-5 space-y-3 animate-pop">
                <div className="flex flex-wrap items-center justify-between gap-2">
                  <div>
                    <h2 className="text-xs sm:text-sm font-extrabold text-slate-900">
                      🎯 Темы с пробелами — стоит повторить перед СОР ({gapTopics.length})
                    </h2>
                    <p className="text-[11px] text-slate-500">
                      Нажми на тему, чтобы подсветить столбец в журнале и увидеть список учеников ниже 60%.
                    </p>
                  </div>
                  {focusedTopicId && (
                    <button
                      type="button"
                      onClick={() => setFocusedTopicId(null)}
                      className="px-2.5 py-1 rounded-lg bg-slate-100 hover:bg-slate-200 text-[11px] font-bold text-slate-600"
                    >
                      Снять выделение
                    </button>
                  )}
                </div>

                <div className="flex flex-wrap gap-2">
                  {gapTopics.map((item) => {
                    const isFocused = focusedTopicId === item.topic.id;
                    const isCritical = item.level === 'gap';
                    return (
                      <button

'use client';

import { useEffect, useState } from 'react';
import Link from 'next/link';
import QRCode from 'qrcode';

export interface PrintableCard {
  studentId: string;
  fullName: string;
  loginUsername: string;
  pinCode: string;
}

interface Props {
  classroomName: string;
  cards: PrintableCard[];
}

export default function PrintCardsClient({ classroomName, cards }: Props) {
  const [origin, setOrigin] = useState('https://math-platform-pink.vercel.app');
  const [qrMap, setQrMap] = useState<Record<string, string>>({});

  useEffect(() => {
    const currentOrigin = window.location.origin;
    setOrigin(currentOrigin);

    let cancelled = false;
    async function generateAll() {
      const next: Record<string, string> = {};
      for (const card of cards) {
        const url = `${currentOrigin}/login?u=${encodeURIComponent(card.loginUsername)}`;
        try {
          next[card.studentId] = await QRCode.toDataURL(url, {
            width: 140,
            margin: 1,
            color: { dark: '#0F172A', light: '#FFFFFF' },
          });
        } catch {
          // Игнорируем единичную ошибку генерации
        }
      }
      if (!cancelled) setQrMap(next);
    }

    generateAll();
    return () => {
      cancelled = true;
    };
  }, [cards]);

  const cleanHost = origin.replace(/^https?:\/\//, '');

  return (
    <div className="min-h-screen bg-[#F6F7FB] print:bg-white">
      {/* Верхняя панель (скрывается при печати) */}
      <div className="print:hidden sticky top-0 z-30 bg-white/95 backdrop-blur border-b border-slate-200">
        <div className="max-w-5xl mx-auto px-4 h-16 flex flex-wrap items-center justify-between gap-3">
          <div>
            <h1 className="text-base sm:text-lg font-extrabold text-slate-900">
              🖨️ Карточки входа — {classroomName}
            </h1>
            <p className="text-xs text-slate-500">
              Карточек: <strong>{cards.length}</strong> · Формат A4, удобно разрезать и вклеить в тетрадь
            </p>
          </div>
          <div className="flex items-center gap-2">
            <Link
              href="/teacher/journal"
              className="px-3.5 py-2 rounded-xl bg-slate-100 hover:bg-slate-200 text-xs font-bold text-slate-700"
            >
              ← В журнал
            </Link>
            <button
              type="button"
              onClick={() => window.print()}
              disabled={cards.length === 0}
              className="btn-tactile px-4 py-2 rounded-xl bg-indigo-600 hover:bg-indigo-700 border-indigo-800 disabled:opacity-50 text-white text-xs sm:text-sm font-extrabold"
            >
              🖨️ Распечатать / Сохранить в PDF
            </button>
          </div>
        </div>
      </div>

'use client';

import { useMemo, useState } from 'react';
import Link from 'next/link';
import MathMarkdown from '@/components/lesson/MathMarkdown';
import { playSound } from '@/lib/audio/sound';

export interface ReviewClassroom {
  id: string;
  name: string;
}

export interface ReviewSubmissionItem {
  id: string;
  studentId: string;
  studentName: string;
  classroomId: string | null;
  classroomName: string | null;
  topicTitle: string;
  topicOrder: number;
  questionMd: string;
  maxPoints: number;
  answerText: string;
  photoUrl: string | null;
  status: string;
  score: number;
  teacherComment: string;
  updatedAt: string;
}

interface AiReviewDraft {
  transcribed_solution: string;
  verdict: 'correct' | 'partial' | 'incorrect' | 'unreadable';
  suggested_score: number;
  error_step: string;
  teacher_comment: string;
}

const VERDICT_META: Record<
  AiReviewDraft['verdict'],
  { label: string; cls: string }
> = {
  correct: {
    label: '✅ Решение верное',
    cls: 'bg-emerald-100 text-emerald-800 border-emerald-300',
  },
  partial: {
    label: '🟡 Частично верно',
    cls: 'bg-amber-100 text-amber-800 border-amber-300',
  },
  incorrect: {
    label: '❌ Есть ошибка',
    cls: 'bg-rose-100 text-rose-800 border-rose-300',
  },
  unreadable: {
    label: '📷 Фото нечитаемо',
    cls: 'bg-slate-100 text-slate-700 border-slate-300',
  },
};

interface Props {
  classrooms: ReviewClassroom[];
  initialItems: ReviewSubmissionItem[];
}

export default function SubmissionsReviewClient({
  classrooms,
  initialItems,
}: Props) {
  const [items, setItems] = useState<ReviewSubmissionItem[]>(initialItems);
  const [classFilter, setClassFilter] = useState<string>('all');
  const [statusFilter, setStatusFilter] = useState<'pending' | 'graded' | 'all'>(
    'pending'
  );
  const [scoreDrafts, setScoreDrafts] = useState<Record<string, number>>(() => {
    const map: Record<string, number> = {};
    for (const it of initialItems) {
      map[it.id] = it.status === 'graded' ? it.score : it.maxPoints;
    }
    return map;
  });
  const [commentDrafts, setCommentDrafts] = useState<Record<string, string>>(() => {
    const map: Record<string, string> = {};
    for (const it of initialItems) {
      map[it.id] = it.teacherComment || '';
    }
    return map;
  });
  const [savingId, setSavingId] = useState<string | null>(null);

  // Состояние ИИ-помощника проверки фото/текста
  const [aiLoadingId, setAiLoadingId] = useState<string | null>(null);
  const [aiReviews, setAiReviews] = useState<Record<string, AiReviewDraft>>({});
  const [aiErrors, setAiErrors] = useState<Record<string, string>>({});

  const pendingCount = useMemo(
    () => items.filter((i) => i.status === 'pending_review').length,
    [items]
  );

  const visibleItems = useMemo(() => {
    return items.filter((item) => {
      if (classFilter === 'unassigned' && item.classroomId !== null) return false;
      if (
        classFilter !== 'all' &&
        classFilter !== 'unassigned' &&
        item.classroomId !== classFilter
      ) {
        return false;
      }

      if (statusFilter === 'pending' && item.status !== 'pending_review') return false;
      if (statusFilter === 'graded' && item.status !== 'graded') return false;
      return true;
    });
  }, [items, classFilter, statusFilter]);

  async function handleAiReview(item: ReviewSubmissionItem) {
    setAiLoadingId(item.id);
    setAiErrors((prev) => {
      const next = { ...prev };
      delete next[item.id];
      return next;
    });

    try {
      const res = await fetch('/api/teacher/ai-review-submission', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ submissionId: item.id }),
      });
      const data = await res.json();

      if (!res.ok) {
        setAiErrors((prev) => ({
          ...prev,
          [item.id]: data.error || 'Не удалось выполнить ИИ-анализ работы.',
        }));
        return;
      }

      const review = data.review as AiReviewDraft;
      playSound('success');
      setAiReviews((prev) => ({ ...prev, [item.id]: review }));
    } catch {
      setAiErrors((prev) => ({
        ...prev,
        [item.id]: 'Сетевая ошибка при обращении к ИИ.',
      }));
    } finally {
      setAiLoadingId(null);
    }
  }

  function applyAiDraft(item: ReviewSubmissionItem, review: AiReviewDraft) {
    playSound('click');
    const clampedScore = Math.max(
      0,
      Math.min(item.maxPoints, Math.round(review.suggested_score))
    );
    setScoreDrafts((prev) => ({ ...prev, [item.id]: clampedScore }));
    setCommentDrafts((prev) => ({ ...prev, [item.id]: review.teacher_comment }));
  }

  async function handleGrade(item: ReviewSubmissionItem) {
    setSavingId(item.id);
    const nextScore = Math.max(
      0,
      Math.min(item.maxPoints, Number(scoreDrafts[item.id] ?? 0))
    );
    const nextComment = commentDrafts[item.id] ?? '';

    try {
      const res = await fetch('/api/teacher/grade-submission', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          submissionId: item.id,
          score: nextScore,
          teacherComment: nextComment,
        }),
      });
      const data = await res.json();
      if (!res.ok) {
        alert(data.error || 'Не удалось сохранить оценку');
        return;
      }

      playSound('success');
      setItems((prev) =>
        prev.map((row) =>
          row.id === item.id
            ? {
                ...row,
                status: 'graded',
                score: nextScore,
                teacherComment: nextComment,
              }
            : row
        )
      );
    } finally {
      setSavingId(null);
    }
  }

  return (
    <div className="min-h-screen bg-[#F6F7FB]">
      <header className="sticky top-0 z-30 bg-white/95 backdrop-blur border-b border-slate-200">
        <div className="max-w-5xl mx-auto px-4 h-16 flex items-center justify-between gap-3">
          <div>
            <h1 className="text-base sm:text-lg font-extrabold text-slate-900">
              📥 Проверка домашних работ
            </h1>
            <p className="text-xs text-slate-500">
              Ждут проверки: <strong className="text-amber-700">{pendingCount}</strong> ·
              Всего развёрнутых: <strong>{items.length}</strong>
            </p>
          </div>
          <div className="flex items-center gap-2">
            <Link
              href="/teacher/journal"
              className="px-3 py-2 rounded-xl bg-indigo-50 hover:bg-indigo-100 text-indigo-800 border border-indigo-200 text-xs font-bold transition"
            >
              📊 Журнал
            </Link>
            <Link
              href="/today"
              className="px-3.5 py-2 rounded-xl bg-slate-100 hover:bg-slate-200 text-xs font-bold text-slate-700 transition"
            >
              ← Сегодня
            </Link>
          </div>
        </div>
      </header>

      <main className="max-w-5xl mx-auto px-4 py-6 space-y-5">
        {/* Фильтры по классу и статусу */}
        <div className="bg-white rounded-2xl border border-slate-200 p-4 flex flex-wrap items-center justify-between gap-3">
          <div className="flex flex-wrap items-center gap-2">
            <span className="text-xs font-bold text-slate-500">Класс:</span>
            <select
              value={classFilter}
              onChange={(e) => setClassFilter(e.target.value)}
              className="px-3 py-2 text-xs font-bold bg-slate-50 border border-slate-200 rounded-xl"
            >
              <option value="all">Все классы</option>
              {classrooms.map((c) => (
                <option key={c.id} value={c.id}>
                  Класс {c.name}
                </option>
              ))}
              <option value="unassigned">Без класса</option>
            </select>
          </div>

          <div className="inline-flex p-1 bg-slate-100 rounded-xl">
            {(
              [
                ['pending', `⏳ Ждут (${pendingCount})`],
                ['graded', '✓ Проверенные'],
                ['all', 'Все'],
              ] as const
            ).map(([id, label]) => (
              <button
                key={id}
                type="button"
                onClick={() => setStatusFilter(id)}
                className={`px-3 py-1.5 rounded-lg text-xs font-bold transition ${
                  statusFilter === id
                    ? 'bg-indigo-600 text-white'
                    : 'text-slate-600 hover:text-slate-900'
                }`}
              >
                {label}
              </button>
            ))}
          </div>
        </div>

        {/* Список работ */}
        {visibleItems.length === 0 ? (

            <div className="text-3xl">🎉</div>
            <p className="text-sm font-extrabold text-slate-800">
              {statusFilter === 'pending'
                ? 'Все работы в этом фильтре уже проверены!'
                : 'Работ по выбранному фильтру пока нет.'}
            </p>
          </div>
        ) : (
          <div className="space-y-4">
            {visibleItems.map((item) => {
              const isGraded = item.status === 'graded';
              const busy = savingId === item.id;
              const aiLoading = aiLoadingId === item.id;
              const aiReview = aiReviews[item.id];
              const aiError = aiErrors[item.id];
              const verdictInfo = aiReview ? VERDICT_META[aiReview.verdict] : null;
              const canAnalyze = Boolean(item.photoUrl || item.answerText.trim());

              return (
                <div
                  key={item.id}
                  className={`bg-white rounded-3xl border-2 p-5 space-y-4 shadow-xs ${
                    isGraded ? 'border-emerald-200' : 'border-amber-200'

                >
                  <div className="flex flex-wrap items-start justify-between gap-2">

interface LastLessonSnapshot {
  courseId: string;
  courseTitle: string;
  pathname: string;
  topicId: string;
  topicTitle: string;
  orderIndex: number;
  quarter: number;
  tab: 'theory' | 'interactive' | 'homework';
  updatedAt: number;
}

interface Props {
  profile: Profile | null;

  teacherClassrooms: ClassroomOption[];

  stats: {
    completedTopics: number;
    simSolved: number;
    bestStreak: number;
  };
  todayIso: string;
  tomorrowIso: string;
}

const LAST_LESSON_STORAGE_KEY = 'mathhub_last_lesson';

const TAB_LABELS: Record<LastLessonSnapshot['tab'], string> = {
  theory: '📘 Слайды и конспект',
  interactive: '🎮 Тренажёр',
  homework: '📝 Домашнее задание',
};

  { id: 'percent_grid', icon: '💯', name: 'Проценты 10×10', desc: 'Сетка и дроби' },
  { id: 'diagrams', icon: '📊', name: 'Диаграммы', desc: 'Круги и столбцы' },
];

function subscribeLastLesson(callback: () => void): () => void {
  if (typeof window === 'undefined') return () => {};
  window.addEventListener('storage', callback);
  return () => window.removeEventListener('storage', callback);
}

function readLastLessonRaw(): string | null {
  if (typeof window === 'undefined') return null;
  try {
    return window.localStorage.getItem(LAST_LESSON_STORAGE_KEY);
  } catch {
    return null;
  }
}

function parseLastLesson(raw: string | null): LastLessonSnapshot | null {
  if (!raw) return null;
  try {
    const parsed = JSON.parse(raw) as Partial<LastLessonSnapshot>;
    if (
      !parsed ||
      typeof parsed.pathname !== 'string' ||
      !parsed.pathname.startsWith('/courses/') ||
      typeof parsed.topicId !== 'string' ||
      typeof parsed.topicTitle !== 'string'
    ) {
      return null;
    }
    const safeTab: LastLessonSnapshot['tab'] =
      parsed.tab === 'interactive' || parsed.tab === 'homework'
        ? parsed.tab
        : 'theory';
    return {
      courseId: parsed.courseId || '',
      courseTitle: parsed.courseTitle || 'Курс математики',
      pathname: parsed.pathname,
      topicId: parsed.topicId,
      topicTitle: parsed.topicTitle,
      orderIndex: Number(parsed.orderIndex) || 1,
      quarter: Number(parsed.quarter) || 1,
      tab: safeTab,
      updatedAt: Number(parsed.updatedAt) || 0,
    };
  } catch {
    return null;
  }
}

'use client';

import { useEffect, useRef } from 'react';

interface Props {
  /** Увеличивай число (1, 2, 3...), чтобы запустить новый залп конфетти */
  burstKey: number;
}

interface Particle {
  x: number;
  y: number;
  vx: number;
  vy: number;
  size: number;
  color: string;
  rotation: number;
  vRot: number;
  life: number;
}

const COLORS = [
  '#6366F1',
  '#10B981',
  '#F59E0B',
  '#EC4899',
  '#0EA5E9',
  '#8B5CF6',
];

export default function ConfettiBurst({ burstKey }: Props) {
  const canvasRef = useRef<HTMLCanvasElement | null>(null);

  useEffect(() => {
    if (burstKey <= 0) return;
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext('2d');
    if (!ctx) return;

    const width = window.innerWidth;
    const height = window.innerHeight;
    canvas.width = width;
    canvas.height = height;

    const particles: Particle[] = Array.from({ length: 85 }, () => {
      const fromLeft = Math.random() < 0.5;
      const startX = fromLeft
        ? width * (0.15 + Math.random() * 0.25)
        : width * (0.6 + Math.random() * 0.25);
      const startY = height * (0.25 + Math.random() * 0.2);

      return {
        x: startX,
        y: startY,
        vx: (Math.random() - 0.5) * 11,
        vy: -(5 + Math.random() * 8),
        size: 6 + Math.random() * 6,
        color: COLORS[Math.floor(Math.random() * COLORS.length)],
        rotation: Math.random() * Math.PI * 2,
        vRot: (Math.random() - 0.5) * 0.25,
        life: 1,
      };
    });

    let rafId = 0;
    const startTime = performance.now();

    function frame(now: number) {
      if (!ctx) return;
      const elapsed = now - startTime;
      ctx.clearRect(0, 0, width, height);

      let alive = 0;
      for (const p of particles) {
        p.x += p.vx;
        p.y += p.vy;
        p.vy += 0.28; // гравитация
        p.vx *= 0.99;
        p.rotation += p.vRot;
        p.life = Math.max(0, 1 - elapsed / 2300);

        if (p.life > 0.02 && p.y < height + 30) {
          alive++;
          ctx.save();
          ctx.globalAlpha = p.life;
          ctx.translate(p.x, p.y);
          ctx.rotate(p.rotation);
          ctx.fillStyle = p.color;
          ctx.fillRect(-p.size / 2, -p.size / 3, p.size, p.size * 0.65);
          ctx.restore();
        }
      }

      if (alive > 0 && elapsed < 2400) {
        rafId = requestAnimationFrame(frame);
      } else {
        ctx.clearRect(0, 0, width, height);
      }
    }

    rafId = requestAnimationFrame(frame);
    return () => cancelAnimationFrame(rafId);
  }, [burstKey]);

  if (burstKey <= 0) return null;

  return (
    <canvas
      ref={canvasRef}
      aria-hidden="true"
      className="fixed inset-0 z-[70] pointer-events-none"
    />
  );
}

'use client';

import { useCallback, useEffect, useId, useRef, useState } from 'react';
import { playSound } from '@/lib/audio/sound';
import {
  closeMathKeyboard,
  openMathKeyboard,
  syncMathKeyboard,
} from '@/lib/keyboard/mathKeyboardStore';

interface Props {
  value: string;
  onChange: (next: string) => void;
  onSubmit?: () => void;
  placeholder?: string;
  disabled?: boolean;
  examMode?: boolean;
  submitLabel?: string;
  className?: string;
}

export default function MathInput({
  value,
  onChange,
  onSubmit,
  placeholder = 'Введи ответ...',
  disabled = false,
  examMode = false,
  submitLabel = '✓ Проверить',
  className = '',
}: Props) {
  const id = useId();
  const inputRef = useRef<HTMLInputElement | null>(null);
  const [preferNative, setPreferNative] = useState(false);

  const handleSwitchNative = useCallback(() => {
    setPreferNative(true);
    closeMathKeyboard(id);
    const el = inputRef.current;
    if (!el) return;
    el.blur();
    setTimeout(() => {
      el.focus();
    }, 60);
  }, [id]);

  // Синхронизируем внешние изменения (например, сброс поля при следующем примере)
  useEffect(() => {
    syncMathKeyboard(id, {
      value,
      placeholder,
      submitLabel,
      examMode,
      disabled,
      onChange,
      onSubmit,
      onSwitchNative: handleSwitchNative,
      inputRef,
    });
  }, [
    id,
    value,
    placeholder,
    submitLabel,
    examMode,
    disabled,
    onChange,
    onSubmit,
    handleSwitchNative,
  ]);

  useEffect(() => {
    return () => closeMathKeyboard(id);
  }, [id]);

  function openDock() {
    if (disabled) return;
    openMathKeyboard({
      id,
      value,
      placeholder,
      submitLabel,
      examMode,
      disabled,
      onChange,
      onSubmit,
      onSwitchNative: handleSwitchNative,
      inputRef,
    });
    // Плавно поднимаем поле, чтобы нижняя клавиатура его не перекрыла
    setTimeout(() => {
      inputRef.current?.scrollIntoView({ behavior: 'smooth', block: 'center' });
    }, 80);
  }

  return (
    <div data-math-input="true" className="relative flex-1 min-w-0">
      <input
        ref={inputRef}
        type="text"
        autoComplete="off"
        autoCorrect="off"
        autoCapitalize="none"
        spellCheck={false}
        inputMode={preferNative ? 'text' : 'none'}
        disabled={disabled}
        value={value}
        placeholder={placeholder}
        onFocus={() => {
          if (!preferNative) openDock();
        }}
        onClick={() => {
          if (!preferNative) openDock();
        }}
        onChange={(e) => onChange(e.target.value)}
        className={`w-full pr-11 ${className}`}
      />

      {!disabled && (
        <button
          type="button"
          title="Открыть математическую клавиатуру MathHub"
          onPointerDown={(e) => e.preventDefault()}
          onClick={() => {
            playSound('click');
            setPreferNative(false);
            inputRef.current?.focus();
            openDock();
          }}
          className="absolute right-2 top-1/2 -translate-y-1/2 w-8 h-8 rounded-lg bg-indigo-50 hover:bg-indigo-100 border border-indigo-200/80 text-indigo-700 flex items-center justify-center text-sm transition active:scale-95"
        >
          ⌨️
        </button>
      )}
    </div>
  );
}

      if (e.key === 'Escape') {

      const target = e.target as HTMLElement | null;
      if (!target) return;
      if (target.closest('[data-math-keyboard]')) return;
      if (target.closest('[data-math-input]')) return;

    const cur = getMathKeyboardSnapshot();

      typeof el.selectionStart === 'number' &&
      typeof el.selectionEnd === 'number'

      const end = el.selectionEnd;

        el.focus();
        const pos = start + text.length;

      typeof el.selectionEnd === 'number'

  const [stats, setStats] = useState<StudentGamificationStats | null>(null);
  const [starsMap, setStarsMap] = useState<Record<string, TopicStarProgress>>({});
  const [rewardBanner, setRewardBanner] = useState<string | null>(null);
  const [loaded, setLoaded] = useState(false);

  const bannerTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
  const starsMapRef = useRef<Record<string, TopicStarProgress>>({});

  useEffect(() => {
    starsMapRef.current = starsMap;
  }, [starsMap]);

  const showBanner = useCallback((message: string) => {
    setRewardBanner(message);
    if (bannerTimerRef.current) clearTimeout(bannerTimerRef.current);
    bannerTimerRef.current = setTimeout(() => {
      setRewardBanner(null);
    }, 5000);
  }, []);

  useEffect(() => {
    return () => {
      if (bannerTimerRef.current) clearTimeout(bannerTimerRef.current);
    };
  }, []);

  useEffect(() => {
    if (!enabled) return;

    let cancelled = false;

    async function load() {
      try {
        const res = await fetch('/api/student/gamification', {
          cache: 'no-store',
        });
        if (!res.ok) return;
        const data = await res.json();
        if (cancelled) return;
        if (data.stats) setStats(data.stats);
        if (data.starsMap) setStarsMap(data.starsMap);
      } catch (err) {
        console.error('load gamification error:', err);
      } finally {
        if (!cancelled) setLoaded(true);
      }
    }

    load();
    return () => {
      cancelled = true;
    };
  }, [enabled]);

  const recordAction = useCallback(
    async (
      topicId: string,
      action: ActionType,
      extra?: { solvedDelta?: number; solvedTotal?: number }
    ) => {
      if (!enabled || !topicId) return;

      const current = starsMapRef.current[topicId];
      if (action === 'slides_done' && current?.slides_done) return;
      if (action === 'homework_done' && current?.homework_done) return;

      try {
        const res = await fetch('/api/student/gamification', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({
            topicId,
            action,
            ...extra,
          }),
        });
        if (!res.ok) return;

        const data = (await res.json()) as {
          stats?: StudentGamificationStats;
          topicStars?: TopicStarProgress;
          awardedXp?: number;
          newBadges?: BadgeDef[];
        };

        if (data.stats) setStats(data.stats);
        if (data.topicStars) {
          setStarsMap((prev) => ({
            ...prev,
            [topicId]: data.topicStars!,
          }));
        }

        const xpGain = Number(data.awardedXp || 0);
        const badges = Array.isArray(data.newBadges) ? data.newBadges : [];

        if (xpGain > 0 || badges.length > 0) {
          playSound('celebrate');
          const parts: string[] = [];
          if (xpGain > 0) parts.push(`+${xpGain} XP`);
          if (badges.length > 0) {
            parts.push(
              `Новая награда: ${badges.map((b) => `${b.icon} ${b.title}`).join(', ')}`
            );
          }
          showBanner(`🎉 ${parts.join(' · ')}`);
        }
      } catch (err) {
        console.error('recordAction error:', err);
      }
    },
    [enabled, showBanner]
  );

  const markSlidesDone = useCallback(
    (topicId: string) => recordAction(topicId, 'slides_done'),
    [recordAction]
  );

  const reportPracticeSolved = useCallback(
    (topicId: string, solvedTotal?: number) =>
      recordAction(
        topicId,
        'practice_solved',
        solvedTotal !== undefined ? { solvedTotal } : { solvedDelta: 1 }
      ),
    [recordAction]
  );

  const markHomeworkDone = useCallback(
    (topicId: string) => recordAction(topicId, 'homework_done'),
    [recordAction]
  );

  return {
    stats,
    starsMap,
    rewardBanner,
    loading: enabled && !loaded,
    markSlidesDone,
    reportPracticeSolved,
    markHomeworkDone,
  };
}

'use client';

export type SoundKind = 'success' | 'error' | 'celebrate' | 'click';

const STORAGE_KEY = 'mathhub_sound';
let sharedCtx: AudioContext | null = null;

function getAudioContext(): AudioContext | null {
  if (typeof window === 'undefined') return null;
  const AudioCtx =
    window.AudioContext ||
    (window as unknown as { webkitAudioContext?: typeof AudioContext }).webkitAudioContext;
  if (!AudioCtx) return null;
  if (!sharedCtx) {
    sharedCtx = new AudioCtx();
  }
  if (sharedCtx.state === 'suspended') {
    sharedCtx.resume().catch(() => {});
  }
  return sharedCtx;
}

export function isSoundEnabled(): boolean {
  if (typeof window === 'undefined') return true;
  return window.localStorage.getItem(STORAGE_KEY) !== 'off';
}

export function setSoundEnabled(enabled: boolean): void {
  if (typeof window === 'undefined') return;
  window.localStorage.setItem(STORAGE_KEY, enabled ? 'on' : 'off');
  window.dispatchEvent(new Event('mathhub-sound-change'));
}

function playTone(
  ctx: AudioContext,
  freq: number,
  startOffset: number,
  duration: number,
  volume: number,
  type: OscillatorType = 'sine',
  endFreq?: number
) {
  const now = ctx.currentTime + startOffset;
  const osc = ctx.createOscillator();
  const gain = ctx.createGain();

  osc.type = type;
  osc.frequency.setValueAtTime(freq, now);
  if (endFreq !== undefined) {
    osc.frequency.exponentialRampToValueAtTime(endFreq, now + duration);
  }

  gain.gain.setValueAtTime(0.0001, now);
  gain.gain.linearRampToValueAtTime(volume, now + 0.015);
  gain.gain.exponentialRampToValueAtTime(0.0001, now + duration);

  osc.connect(gain);
  gain.connect(ctx.destination);
  osc.start(now);
  osc.stop(now + duration + 0.02);
}

export function playSound(kind: SoundKind): void {
  if (typeof window === 'undefined' || !isSoundEnabled()) return;

  try {
    const ctx = getAudioContext();
    if (!ctx) return;

    if (kind === 'success') {
      // Тёплый звон маримбы вверх (C5 -> E5)
      playTone(ctx, 523.25, 0, 0.14, 0.12, 'triangle');
      playTone(ctx, 659.25, 0.09, 0.22, 0.14, 'triangle');
      navigator.vibrate?.(15);
    } else if (kind === 'celebrate') {
      // Приятный аккорд-перелив (C5 -> E5 -> G5 -> C6)
      playTone(ctx, 523.25, 0, 0.16, 0.11, 'triangle');
      playTone(ctx, 659.25, 0.08, 0.16, 0.12, 'triangle');
      playTone(ctx, 783.99, 0.16, 0.18, 0.12, 'triangle');
      playTone(ctx, 1046.5, 0.24, 0.32, 0.14, 'sine');
      navigator.vibrate?.([20, 40, 25]);
    } else if (kind === 'error') {
      // Мягкий тихий «тук» без резкого негатива
      playTone(ctx, 210, 0, 0.14, 0.08, 'sine', 165);
    } else if (kind === 'click') {
      // Лёгкий щелчок
      playTone(ctx, 580, 0, 0.04, 0.04, 'sine');
    }
  } catch {
    // Игнорируем ограничения автовоспроизведения браузера
  }
}

export function transliterateToSlug(input: string): string {
  return input
    .toLowerCase()
    .split('')
    .map((ch) => CYRILLIC_MAP[ch] ?? ch)
    .join('')
    .replace(/[^a-z0-9]/g, '');
}

    .replace(/^\s*\d+[\.\)\-\s]+/, '')
    .trim();
  const parts = cleaned.split(/\s+/).filter(Boolean);

export function getLocalDateISO(offsetDays = 0): string {
  const d = new Date();
  d.setDate(d.getDate() + offsetDays);
  const year = d.getFullYear();
  const month = String(d.getMonth() + 1).padStart(2, '0');
  const day = String(d.getDate()).padStart(2, '0');
  return `${year}-${month}-${day}`;
}

export function formatShortRuDate(isoDate: string): string {
  const [y, m, d] = isoDate.split('-').map(Number);
  if (!y || !m || !d) return isoDate;
  const date = new Date(y, m - 1, d);
  return date.toLocaleDateString('ru-RU', { day: 'numeric', month: 'short' });
}

export function getScheduleBadge(
  lessonDate: string | null | undefined
): 'today' | 'tomorrow' | 'upcoming' | 'past' | null {
  if (!lessonDate) return null;
  const today = getLocalDateISO(0);
  const tomorrow = getLocalDateISO(1);
  if (lessonDate === today) return 'today';
  if (lessonDate === tomorrow) return 'tomorrow';
  if (lessonDate > tomorrow) return 'upcoming';
  return 'past';
}

export function getLevelInfo(xp: number) {
  const safeXp = Math.max(0, Number(xp) || 0);
  let current = LEVELS[0];
  let next: (typeof LEVELS)[number] | null = LEVELS[1];

  for (let i = 0; i < LEVELS.length; i++) {
    if (safeXp >= LEVELS[i].minXp) {
      current = LEVELS[i];
      next = LEVELS[i + 1] ?? null;
    }
  }

  const progressPct = next
    ? Math.min(
        100,
        Math.round(((safeXp - current.minXp) / (next.minXp - current.minXp)) * 100)
      )
    : 100;

  return {
    ...current,
    nextLevel: next,
    progressPct,
    xpToNext: next ? Math.max(0, next.minXp - safeXp) : 0,
  };
}

  return () => listeners.delete(listener);
}

export function syncMathKeyboard(
  id: string,
  patch: Partial<Omit<MathKeyboardConfig, 'id'>>
): void {
  if (!snapshot || snapshot.id !== id) return;

  const uiChanged =
    (patch.value !== undefined && patch.value !== snapshot.value) ||
    (patch.placeholder !== undefined && patch.placeholder !== snapshot.placeholder) ||
    (patch.submitLabel !== undefined && patch.submitLabel !== snapshot.submitLabel) ||
    (patch.examMode !== undefined && patch.examMode !== snapshot.examMode) ||
    (patch.disabled !== undefined && patch.disabled !== snapshot.disabled);

  snapshot = { ...snapshot, ...patch };
  if (uiChanged) {
    emit();
  }
}

  if (!snapshot || snapshot.id !== id) return;
  if (snapshot.value === nextValue) return;
  snapshot = { ...snapshot, value: nextValue };
  emit();
}

  if (!snapshot) return;
  if (id && snapshot.id !== id) return;
  snapshot = null;
  emit();

export function gcd(a: number, b: number): number {
  let x = Math.abs(a);
  let y = Math.abs(b);
  while (y) {
    const t = y;
    y = x % y;
    x = t;
  }
  return x || 1;
}

export function lcm(a: number, b: number): number {
  return Math.abs(a * b) / gcd(a, b);
}

export function randInt(min: number, max: number): number {
  return Math.floor(Math.random() * (max - min + 1)) + min;
}

'use client';

export async function compressImageFile(
  file: File,
  maxDimension = 1400,
  quality = 0.78
): Promise<File> {
  if (!file.type.startsWith('image/')) return file;

  return new Promise((resolve) => {
    const objectUrl = URL.createObjectURL(file);
    const img = new Image();

    img.onload = () => {
      try {
        let { width, height } = img;
        if (width > maxDimension || height > maxDimension) {
          if (width >= height) {
            height = Math.round((height * maxDimension) / width);
            width = maxDimension;
          } else {
            width = Math.round((width * maxDimension) / height);
            height = maxDimension;
          }
        }

        const canvas = document.createElement('canvas');
        canvas.width = width;
        canvas.height = height;
        const ctx = canvas.getContext('2d');

        if (!ctx) {
          URL.revokeObjectURL(objectUrl);
          resolve(file);
          return;
        }

        ctx.fillStyle = '#FFFFFF';
        ctx.fillRect(0, 0, width, height);
        ctx.drawImage(img, 0, 0, width, height);

        canvas.toBlob(
          (blob) => {
            URL.revokeObjectURL(objectUrl);
            if (!blob || blob.size >= file.size) {
              resolve(file);
              return;
            }
            const baseName = file.name.replace(/\.[^.]+$/, '') || 'notebook';
            resolve(new File([blob], `${baseName}.jpg`, { type: 'image/jpeg' }));
          },
          'image/jpeg',
          quality
        );
      } catch {
        URL.revokeObjectURL(objectUrl);
        resolve(file);
      }
    };

    img.onerror = () => {
      URL.revokeObjectURL(objectUrl);
      resolve(file);
    };

    img.src = objectUrl;
  });
}

export function applyTheme(theme: AppTheme): void {
  if (typeof window === 'undefined') return;
  document.documentElement.setAttribute('data-theme', theme);
  try {
    window.localStorage.setItem(STORAGE_KEY, theme);
  } catch {
    // Игнорируем ограничения приватного режима
  }
  window.dispatchEvent(new Event(THEME_EVENT));
}

export function subscribeTheme(callback: () => void): () => void {
  if (typeof window === 'undefined') return () => {};
  window.addEventListener(THEME_EVENT, callback);
  window.addEventListener('storage', callback);
  return () => {
    window.removeEventListener(THEME_EVENT, callback);
    window.removeEventListener('storage', callback);
  };
}

import { createServerClient } from '@supabase/ssr';
import { NextResponse, type NextRequest } from 'next/server';

export async function middleware(request: NextRequest) {
  if (request.nextUrl.pathname === '/') {
    return NextResponse.redirect(new URL('/today', request.url));
  }
  let response = NextResponse.next({ request });

  const supabase = createServerClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!,
    {
      cookies: {
        getAll: () => request.cookies.getAll(),
        setAll: (cookiesToSet) => {
          cookiesToSet.forEach(({ name, value }) => request.cookies.set(name, value));
          response = NextResponse.next({ request });
          cookiesToSet.forEach(({ name, value, options }) =>
            response.cookies.set(name, value, options)
          );
        },
      },
    }
  );

  await supabase.auth.getUser();
  return response;
}

  ],
